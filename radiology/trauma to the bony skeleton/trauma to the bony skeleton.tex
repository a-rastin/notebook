\section*{\en{Trauma to the bony skeleton}}
\begin{enumerate}[label=\arabic*.,leftmargin=*,itemsep=.6em]
\item \en{If the cortex is broken through and through, the fracture is called complete.}
\begin{latin}\begin{itemize}[nosep]
\item Otherwise incomplete.
\end{itemize}\end{latin}
\item \en{Incomplete fracture}
\begin{latin}\begin{itemize}[nosep]
\item In bones that are softer then normal
\begin{itemize}[nosep]
\item In children or in adults with bone-softening disease.
\item Paget disease
\end{itemize}
\item greenstick fracture, torus fracture (buckle fracture)
\end{itemize}\end{latin}
\item یافته‌های رادیولوژیک شکستگی حاد
\begin{latin}\begin{itemize}[nosep]
\item Fracture lines tend to be more lucent then normal lines
\item Abrupt discontinuity of the cortex
\item The edge of the fracture may be jagged or irregular
\end{itemize}\end{latin}
\item \en{Buckle fracture $\longrightarrow$ bulging of the cortex}
\item شکستگی‌های قدیمی بهبود نیافته ممکن است شبیه شکستگی حاد دیده شوند.
\item استخوان‌های \en{sesamoid} معمولاً دوطرفه و قرینه هستند.
\item مکان‌های شایع برای دیدن استخوان \en{sesamoid}
\begin{latin}\begin{itemize}[nosep]
\item Thumb
\item Great toe
\item Posterolateral aspect of the knee (Fabella)
\end{itemize}\end{latin}
\end{enumerate}

\begin{enumerate}[label=\arabic*.,leftmargin=*,itemsep=.6em]\setcounter{enumi}{7}
\item دررفتگی (دیسلوکیشن)
\begin{itemize}[nosep]
\item فقط در مفاصل رخ می‌دهد.
\item دو جزء استخوانی مفصل دیگر با هم در تماس نیستند.
\end{itemize}
\item نیمه دررفتگی (ساب لاکسیشن)
\begin{itemize}[nosep]
\item دو جزء استخوانی در تماس نسبی هستند.
\item فقط در مفاصل رخ می‌دهد.
\end{itemize}
\item مقایسه دررفتگی شانه و هیپ
\begin{itemize}[itemsep=.3em]
\item جهت
\begin{itemize}[nosep]
\item شانه: قدامی، ساب کوراکوئید
\item هیپ: خلفی و فوقانی
\end{itemize}
\item علت
\begin{itemize}[nosep]
\item شانه: ابداکشن + اکسترنال روتیشن + اکستنشن
\item هیپ: برخورد زانو به داشبورد ماشین
\end{itemize}
\item همراهی
\begin{itemize}[nosep]
\item شانه: شکستگی سر هومروس \TBSarrow\ \en{Hill-Sachs deformity}؛ شکستگی گلنوئید \TBSarrow\ \en{Bankart fracture}
\item هیپ: شکستگی \en{posterior rim of the acetabulum}
\end{itemize}
\end{itemize}
\item توصیف شکستگی‌ها
\begin{latin}\begin{enumerate}[label=\alph*),itemsep=.3em]
\item Number of fracture fragments $\longrightarrow$ Simple; Comminuted
\item Direction of fracture line $\longrightarrow$ Transverse; Oblique (diagonal); Spiral
\item Relation of one fragment to another $\longrightarrow$ Displacement; Angulation, Shortening, Rotation
\item Open to the atmosphere $\longrightarrow$ Closed; Open
\end{enumerate}\end{latin}
\end{enumerate}

\begin{enumerate}[label=\arabic*.,leftmargin=*,itemsep=.6em]\setcounter{enumi}{11}
\item \en{Comminuted fracture}
\begin{latin}\begin{itemize}[itemsep=.3em]
\item Segmental frac. $\longrightarrow$ A portion of the shaft exists as an isolated fragment
\item Butterfly fragment $\longrightarrow$ The central fragment has a triangular shape
\end{itemize}\end{latin}
\item شکستگی‌های عرضی \TBSarrow\ نیرو عمود بر محور شفت استخوان وارد شده است.
\item شکستگی ابلیک \TBSarrow\ نیرو هم‌جهت با محور شفت استخوان وارد شده است.
\item شکستگی اسپایرال
\begin{itemize}[nosep]
\item هنگام دویدن پا داخل یک چاله بود.
\item عموماً ناپایدار و همراه با آسیب بافت نرم اطراف مثل لیگامنت‌ها، تاندون‌ها هستند.
\end{itemize}
\item توصیف \en{displacement} \TBSarrow\ میزانی که بخش دیستال شکستگی \en{off-set} شده است.
\begin{latin}Front-to-back and side-to-side from the proximal fragment\end{latin}
\item \en{Angulation} \TBSarrow\ زاویه بین بخش دیستال و پروگزیمال شکستگی
\item شکستگی باز با مشاهده و با بالین بیمار مشخص می‌شود و نه گرافی
\begin{itemize}[nosep]
\item ریسک استئومیلیت بالاتر
\end{itemize}
\item \en{Avulsion fracture}
\begin{itemize}[nosep]
\item زمانی که بخش جدا شدهٔ استخوان توسط تاندون کشیده شده و از جای خود خارج می‌شود.
\end{itemize}
\end{enumerate}

\begin{enumerate}[label=\arabic*.,leftmargin=*,itemsep=.6em]\setcounter{enumi}{23}
\item \en{Colles fracture}
\begin{latin}\begin{itemize}[nosep]
\item Fracture of the distal radius with dorsal angulation.
\item Fall on the outstretched hand
\item Frequently an associated fracture of the ulnar styloid
\end{itemize}\end{latin}
\item \en{Smith fracture}
\begin{latin}\begin{itemize}[nosep]
\item Fracture of the distal radius with palmar angulation.
\item Is a reverse Colles fracture.
\item Fall on the back of the flexed hand.
\end{itemize}\end{latin}
\item \en{Jones fracture}
\begin{latin}\begin{itemize}[nosep]
\item Transverse fracture of the 5th metatarsal about 1 to 2 cm from it's base.
\item Plantar flexion of the foot and inversion of the ankle.
\end{itemize}\end{latin}
\item \en{Boxer's fracture}
\begin{latin}\begin{itemize}[nosep]
\item Fracture of the neck of the 5th metacarpal with palmar angulation.
\item Sometimes the 4th metacarpal may be involved.
\item Result of punching a person or wall.
\end{itemize}\end{latin}
\item \en{March fracture}
\begin{latin}\begin{itemize}[nosep]
\item Is a type of stress fracture.
\item Most often affecting the shaft of the second and third metatarsal.
\end{itemize}\end{latin}
\item \en{Lisfranc-fracture-dislocation}
\begin{latin}\begin{itemize}[nosep]
\item Relatively uncommon
\item CT or MRI may be needed
\item Athletic injuries or motor vehicle accidents
\end{itemize}\end{latin}
\end{enumerate}
