\CFpage{۱}
\subsection*{ژنتیک و پاتوفیزیولوژی}
\begin{itemize}
\item \en{CF}: وراثت \en{AR}؛ بازوی بلند کروموزوم ۷، ژن \en{CFTR} (کانال کلر).
\end{itemize}
\begin{latin}
Cystic Fibrosis transmembrane conductance regulator
\end{latin}
\begin{itemize}
\item در \en{CF} سطح سدیم و کلر به‌شدت بالاست در عرق: \en{Sweat Cl test}.
\item تست کلر عرق در ۹۹\% موارد \en{CF} مثبت می‌شود: کلر عرق \en{$\geq 60\,\mathrm{mEq/L}$}.
\item ترشحات غیرطبیعی در راه‌های هوایی، زمینه برای کلونیزه شدن باکتری‌ها ایجاد می‌کند؛ سپس \en{Chronic airway infections} و برونشکتازی.
\item \en{CF} یک بیماری مزمن و پیشرونده است.
\end{itemize}

\subsection*{تظاهرات}
\begin{itemize}
\item سوءجذب پروتئین و چربی: \en{FTT}، هایپوآلبومینوری، استئاتوره.
% The source reads «هایپوآلبومینوری» here; page 2 writes «هایپوپروتئینمی».
\item بیماری کبدی: فیبروز کولانژیتیک.
\item عفونت مزمن تنفسی.
\item تظاهرات تنفسی: سرفهٔ مزمن، آسم غیرقابل کنترل، عفونت مزمن تنفسی.
\item سایر تظاهرات: پانکراتیت راجعه، پولیپ بینی، سینوزیت مزمن.
\item \textbf{کلابینگ انگشتان در \en{CF} شایع است حتی اگر دچار بیماری ریوی نشده باشند.}
\end{itemize}

\subsection*{عوارض تنفسی}
\begin{itemize}
\item برونشکتازی، برونشیت، برونشیولیت، پنومونی.
\item هموپتزی، پنوموتوراکس.
\item پولیپ بینی، سینوزیت.
\item کورپولمونل (به دلیل هایپوکسی مزمن)، نارسایی تنفسی.
\end{itemize}

\CFpage{۲}
\subsection*{عوارض گوارشی و سایر عوارض}
\begin{itemize}
\item عوارض گوارشی: ایلئوس مکونیوم (در نوزادان)، آترزی روده (نوزادان).
\item نارسایی پانکراس، سوءجذب چربی و ویتامین‌ها.
\item سیروز کبدی، پانکراتیت.
\item دیگر عوارض: ناباروری، بلوغ دیررس، دیابت.
\item آلکالوز متابولیک هایپوناترمیک ــ هایپوکلرمیک: به علت از دست دادن کلر از طریق عرق.
\item پاتوژن‌های شایع عفونت مزمن تنفسی: استاف اورئوس، سودومونا آئروژینوزا.
\end{itemize}

\subsection*{\en{ABPA}}
\begin{itemize}
\item آلرژیک برونکوپولمونری آسپرژیلوزیس.
\item واکنش ازدیاد حساسیت به آسپرژیلوس در راه هوایی بیماران \en{CF}.
\item درمان: \en{CS} سیستمیک (پردنیزون) + ایتراکونازول (ضد قارچ).
\end{itemize}

\subsection*{هموپتزی و درگیری پانکراس}
\begin{itemize}
\item هموپتزی در \en{CF}: خفیف، ناشی از عفونت؛ ماژور، ناشی از آسیب شریان‌های برونشیال.
\item بیش از ۹۰\% بیماران \en{CF} با نارسایی مادرزادی اندوکرین پانکراس متولد می‌شوند.
% «اندوکرین» is retained as written, without correcting the source.
\item سوءجذب به صورت هایپوپروتئینمی و ادم پریفرال در شیرخواران بروز می‌کند.
\item انسداد روده در \en{CF}:
  \begin{itemize}
  \item در بدو تولد: ایلئوس مکونیوم.
  \item کودکان بزرگ‌تر: \en{DIOS}.
  \end{itemize}
\end{itemize}
\begin{latin}
Distal Intestinal Obstruction Syndrome
\end{latin}

\subsection*{تظاهرات تولیدمثلی}
\begin{itemize}
\item در مردان: آزواسپرمی + فقدان مادرزادی واز دفران.
\item در زنان: آمنورهٔ ثانویه؛ زنان \en{CF} می‌توانند باردار شوند.
\end{itemize}

\subsection*{چه کسانی در کودکی باید از نظر \en{CF} بررسی شوند؟}
\begin{itemize}
\item همهٔ نوزادان با تست بدو تولد مثبت.
\item شیرخوار با مکونیوم ایلئوس.
\end{itemize}

\CFpage{۳}
\begin{itemize}
\item در شیرخوار: \en{FTT}، فیبروز کولانژیتیک، علائم مزمن تنفسی، اختلال الکترولیت‌ها.
\item \en{CF} باید در تشخیص افتراقی بیماری کودکان با علائم مزمن تنفسی و گوارشی قرار بگیرد، خصوصاً اگر همراه با کلابینگ باشد.
\item \textbf{هر کودک زیر ۱۲ سال با پولیپ بینی باید از نظر \en{CF} بررسی شود.}
\item خواهر و برادر بیمار \en{CF} باید از نظر بیماری بررسی شوند.
\end{itemize}

\subsection*{تست‌ها و معیارهای تشخیص}
\begin{itemize}
\item تست \en{DNA}: ۲۰۰۰ موتاسیون \en{CF} را بررسی می‌کند؛ ولی منفی کاذب دارد.
\item تست غربالگری نوزاد:
  \begin{itemize}
  \item تست \en{DNA}.
  \item تست \en{IRT}: \en{elevated immunoreactive trypsinogen}.
  \end{itemize}
\item تست اصلی: بررسی کلر عرق.
\item کرایتریا تشخیص \en{CF}:
  \begin{itemize}
  \item وجود حداقل یک تظاهر تیپیک بیماری (گوارشی، تنفسی و \ldots).
  \item دو تست کلر عرق افزایش‌یافته: \en{$\geq 60$} مثبت، \en{30--59} مرزی.
  \item دو موتاسیون شناخته‌شدهٔ \en{CF} با تست ژنتیک.
  \item نازال پوتانسیل دیفرنس تست.
  \end{itemize}
% The handwritten diagnostic arrows are listed without adding AND/OR logic.
\item سطح پایین الاستاز مدفوع: نارسایی اندوکرین پانکراس.
% «اندوکرین» is retained as written.
\end{itemize}

\subsection*{درمان}
\begin{itemize}
\item درمان در \en{CF} صرفاً برای عوارض تنفسی و گوارشی است.
\item مدولاتورهای \en{CFTR}: \en{Ivacaftor}، \en{Lumacaftor}، \en{Tezacaftor}، \en{Elexacaftor}.
\end{itemize}

\subsection*{درمان گوارشی و تغذیه‌ای}
\begin{itemize}
\item درمان نارسایی اگزوکرین پانکراس: با کپسول‌های انتریک کوتد آنزیم‌های پانکراس؛ لیپاز و پروتئاز.
\item رژیم غذایی: پرکالری + مکمل ویتامین.
\item چربی حتی با وجود استئاتوره حذف نشود.
\item حداکثر دوز لیپاز: \en{$2500\,\mathrm{U/kg}$} برای هر وعده.
\item دوز بالاتر: \en{Fibrosing Colonopathy}.
\item درمان ایلئوس مکونیوم در شیرخواران: انما گاستروگرافین.
\end{itemize}
