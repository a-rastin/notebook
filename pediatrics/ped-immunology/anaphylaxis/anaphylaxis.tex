\subsection*{کهیر و آنژیوادم}
\begin{itemize}
  \item حاد \ANarrow{} \textbf{کمتر از ۶ هفته}.
  \item مزمن \ANarrow{} راجعه، اینترمیتنت، روزانه.
  \item فیزیکال \ANarrow{} فشار، سرما، گرما، فعالیت، ویبریشن، خورشید.
  \item مکانیسم \lr{IgE} در کهیر حاد بیشتر است تا مزمن.
  \item شایع‌ترین علت کهیر حاد در اطفال \ANarrow{} \textbf{بیماری‌های ویروسی}.
  \item ۴۰\% موارد کهیر مزمن \ANarrow{} \lr{IgG} وجود دارد که به \lr{IgE} می‌چسبد.
  \item شایع‌ترین کهیر فیزیکال \ANarrow{} \textbf{درماتوگرافیسم}
  \ANarrow{} \ANdeleted{} \textbf{\lr{IgE}} مدیتد.
\end{itemize}

\subsection*{کهیر کولینرژیک}
\begin{itemize}
  \item مدور، \ANrange{1}{3} میلی‌متر با حاشیهٔ اریتماتو.
  \item به دنبال افزایش دمای مرکزی بدن.
  \item در جوانان شایع است.
  \item ورزش، حمام گرم، استرس روانی.
  \item اگر علائم تنفسی داشته باشد \ANarrow{} آنافیلاکسی به دنبال فعالیت.
\end{itemize}

\subsection*{کهیر ناشی از سرما}
\begin{itemize}
  \item چند دقیقه بعد از مواجهه / یا به دنبال گرم شدن.
  \item تورم لب‌ها به دنبال مایعات سرد.

\subsection*{آنژیوادم ارثی \lr{(HAE)}}
\begin{itemize}
  \item وراثت \textbf{\lr{AD}}، \textbf{کمبود مهارکنندهٔ \lr{\(C_1\)} ـ استراز}.
  \item اپیزودهای راجعهٔ غیرقابل پیش‌بینی تورم.
  \begin{itemize}
    \item صورت، اندام‌ها، ژنیتالیا، شکم، فارنکس، اوروفارنکس.
    \item به دنبال تروما.
  \end{itemize}
  \item علت مرگ: آسفیکسی به دنبال تورم لارنژیال.
  \item به ندرت کنار آنژیوادم ممکن است کهیر داشته باشند.
  \item \textbf{تورم با \lr{OCS} یا آنتی‌هیستامین رفع نمی‌شود}.
  \item انواع:
  \begin{itemize}
    \item تیپ \lr{I} \ANarrow{} ۸۵\% بیماران؛ میزان \ANdeleted{} تولید کم است.
    \item تیپ \lr{II} \ANarrow{} ۱۵\% بیماران؛ مهارکننده خراب است.
  \end{itemize}
  \item تست غربالگری اولیه \ANarrow{} سطح \lr{\(C_4\)}
  \lr{\(\downarrow\)} (نرمال).
  % The handwritten parenthesis reads "نرمال"; retained without correction.
  \item در حملهٔ حاد \lr{HAE} \ANarrow{} سطح \lr{\(C_2\)}
  \lr{\(\downarrow\)} است.
  \item بعضی بیماران مهارکنندهٔ نرمال دارند \ANarrow{}
  این دسته از بیماران \ANunclear{بیشتر \lr{F XII}؛ عدد رومی بازنویسی شده است}.
  \item درمان:
  \begin{itemize}
    \item حملهٔ حاد.
    \item پروفیلاکسی \ANarrow{} کنسانترهٔ مهارکنندهٔ \lr{\(C_1\)}.
    \item مهارکننده (بلاک‌کنندهٔ خاص) برادی‌کینین.
  \end{itemize}
\end{itemize}

\subsection*{تشخیص آنافیلاکسی}
شروع حاد علائم با درگیری پوست یا مخاط + یکی از موارد زیر:
\begin{itemize}
  \item دیسترس تنفسی.
  \item کاهش فشار خون.
  \item اختلال \lr{End-organ}.
\end{itemize}

\begin{itemize}
  \item بیماران با آنژیوادم راجعه که کهیر ندارند \ANarrow{} بررسی از نظر \lr{HAE}.
  \item سطح سرمی تریپتاز \ANarrow{} \lr{1--1.5 h} پس از آنافیلاکسی بالا می‌رود.
  \begin{itemize}
    \item سطح \lr{N/L} ردکنندهٔ آنافیلاکسی نیست.
  \end{itemize}
\end{itemize}

\end{itemize}
