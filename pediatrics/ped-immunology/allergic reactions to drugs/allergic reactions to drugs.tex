\subsection*{\lr{Adverse drug reaction (ARD)}}
\begin{latin}
\begin{itemize}
  \item Type A: predictable
  \item Type B: unpredictable
\end{itemize}
\end{latin}

\begin{latin}
\noindent Type A: Drug Toxicity, Drug interactions, Adverse effects
\par\noindent\(\longrightarrow\) Are Dose Dependent
\par\medskip
\noindent Type B \(\longrightarrow\) Generally dose independent
\end{latin}
\begin{itemize}
  \item عموماً یک پره‌دیسپوزیشن وجود دارد.
  \item \lr{Idiosyncratic reactions, Allergic reactions, Pseudoallergic reactions}
\end{itemize}

\begin{itemize}
  \item آنافیلاکسی \ADRarrow{} \lr{Drug specific IgE Ab}.
  \item واکنش‌های پوستی تأخیری \ADRarrow{} \lr{Drug specific T cells}.
  \item واکنش‌های سودوآلرژیک \ADRarrow{} با مکانیسم غیر از \lr{IgE release}،
  بازوفیل و ماست‌سل فعال می‌شوند.
  \par مثال: \lr{Vancomycin induced flushing}.
\end{itemize}

\begin{latin}
\noindent Drug independent cross-reactive Ags
\par\noindent\(\longrightarrow\) Cetuximab-induced anaphylaxis
\end{latin}
از قبل \lr{IgE} علیه گالاکتوز ـ \lr{\(\alpha\)-1}.

\subsection*{دسته‌بندی آلرژی‌های دارویی}
\begin{description}
  \item[تیپ ۱:] \lr{IgE} \ADRarrow{} دگرانولیشن ماست‌سل و ائوزینوفیل
  \ADRarrow{} شوک \textbf{آنافیلاکسی، کهیر، آنژیوادم}، برونکواسپاسم
  \ADRarrow{} ظرف \ADRrange{1}{6} ساعت از مصرف دارو.

  \item[تیپ ۲:] \lr{IgG} و کمپلمان \ADRarrow{} آسیب سلولی ناشی از
  \lr{IgG} و کمپلمان \ADRarrow{} \textbf{سایتوپنی}
  \ADRarrow{} \ADRrange{5}{15} روز پس از شروع دارو.

  \item[تیپ ۳:] \lr{IgM} یا \lr{IgG} + کمپلمان یا \lr{FcR}
  \ADRarrow{} دپوزیشن کمپلکس ایمنی \ADRarrow{} کهیر،
  \textbf{واسکولیت، سرم سیکنس}.
  \par \ADRrange{7}{8} روز از شروع دارو برای کهیر و سرم سیکنس؛
  \ADRrange{7}{21} روز برای واسکولیت.

  \item[تیپ \lr{4a}:] \lr{Th1 (IFN-\(\gamma\))}
  \ADRarrow{} التهاب مونوسیتیک \ADRarrow{} \textbf{اگزما}
  \ADRarrow{} \ADRrange{1}{21} روز بعد از شروع دارو.

  \item[تیپ \lr{4b}:] \lr{Th2 (IL-4, IL-5)}
  \ADRarrow{} التهاب ائوزینوفیلیک \ADRarrow{}
  اگزانتم ماکولوپاپولر، \textbf{\lr{DRESS}}.
  \par ۱ تا چند روز برای \lr{MPE} و \ADRrange{2}{6} هفته برای \lr{DRESS}.

  \item[تیپ \lr{4c}:] \lr{T} سایتوتوکسیک \ADRarrow{} مرگ کراتینوسیت
  \ADRarrow{} اگزانتم ماکولوپاپولر، \textbf{\lr{SJS/TEN}}، اگزانتم پوسچولر.
  \par \ADRrange{1}{2} روزه برای واکنش‌های خفیف؛
  \ADRrange{4}{28} روز برای \lr{SJS/TEN}.
  \par یادداشت حاشیه با شاخه از تیپ \lr{4c}: \lr{fixed drug eruption}.

  \item[تیپ \lr{4d}:] \lr{T cell (IL-8)}
  \ADRarrow{} التهاب نوتروفیلیک \ADRarrow{} اگزانتم پوسچولر ژنرالیزهٔ حاد
  \ADRarrow{} \ADRrange{1}{2} روز بعد از شروع.
\end{description}

\subsection*{تشخیص}
\begin{itemize}
  \item تست پوستی سریع و حساس برای ارزیابی وجود \lr{IgE}.
  \par برای پنی‌سیلین \lr{\(\checkmark\)}، سایر آنتی‌بیوتیک‌ها \lr{\(\times\)}.
  \par منفی شدن آن ردکننده نیست.
  \item تست کومبس مستقیم و غیرمستقیم \ADRarrow{} اغلب به آنتی‌بیوتیک در زمینهٔ دارو مثبت می‌شود.
  \item سطح سرمی تریپتاز \ADRarrow{} به دنبال دگرانولیشن ماست‌سل
  \lr{\(\uparrow\)} می‌رود.
  \item \lr{Patch test} \ADRarrow{} دقیق‌ترین تست برای درماتیت تماسی با داروها.
  \item \lr{Graded drug challenge} \ADRarrow{} اگر \lr{N/L} باشد یعنی فرد به آن دارو
  قطعاً حساسیت ندارد.
\end{itemize}
