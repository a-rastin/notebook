\textbf{سربرگ و حاشیه:} \lr{Nelson 22}؛ ۲۲۱ ـ \lr{Fever in older children}؛
\lr{Subject:} (خالی)؛ \lr{Date: / /}؛ شمارهٔ برگه: ۱؛ \lr{YASHA}.

علل تب در شیرخواران بزرگ‌تر و کودکان: ویروسی / باکتریایی.
\begin{itemize}
\item توسط ارگان درگیر در بدن.
\end{itemize}
\FeverOlderUnclear{سطر دوم مقدمه به صورت «توسط ارگان درگیر در بدن» خوانده می‌شود؛ واژهٔ نخست شبیه «توسم» نیز دیده می‌شود.}

\subsection*{عفونت‌های ویروسی}
\begin{itemize}
\item شایع‌ترین علل تب.
\item الگوی زمانی دارند: انتهای تابستان، ابتدای پاییز: انتروویروس، کوکساکی.
\end{itemize}
\begin{latin}
hand-foot-and-mouth disease, herpangina, Aseptic meningitis
\end{latin}
\begin{itemize}
\item اواخر پاییز و زمستان: \lr{URI} ویروسی، \lr{LRI} ویروسی؛ \lr{RSV}، آنفلوآنزا.
\item \lr{GE}: نوروویروس، روتاویروس.
\item پاییز و بهار: \lr{Croup} (لارنگوتراکئوبرونشایتیس): پاراآنفلوآنزا.
\item بیشتر شیرخواران و نوپایان را تحت تأثیر قرار می‌دهد.
\item واریسلا: فصل شایع زمستان و ابتدای بهار.
\end{itemize}

\subsection*{عفونت‌های باکتریال}
عفونت‌های شایع شامل اوتیت میانی حاد، فارنژیت استرپتوکوکی.

\subsection*{اوتیت میانی حاد}
تشخیص با مشاهدهٔ \lr{bulging}، اریتم و عدم حرکت \lr{TM}.

\subsection*{فارنژیت استرپتوکوکی}
\begin{itemize}
\item بیشتر در اواخر پاییز تا زمستان.
\item قبل از ۳ سالگی ناشایع است.
\end{itemize}

\FeverOlderPage{۲}{02}
\FeverOlderHeader{۲}
یافتهٔ سمعی موضعی شامل کراکل: بیانگر \lr{RTI}ها؛ پنومونی باکتریال.
ولی ممکن است در پنومونی وایرال یا برونشیولیت هم سمع شود.

\subsection*{پنومونی آتیپیکال}
مایکوپلاسما.
\begin{itemize}
\item بیشتر در سن مدرسه و معمولاً همراه با سردرد، گلودرد، \lr{malaise} و تب \lr{low grade}.
\end{itemize}

\subsection*{آبسهٔ رتروفارنژیال}
\begin{itemize}
\item بیشتر خردسال و بچهٔ کم‌سن.
\item درد گردن (خصوصاً حین اکستنشن)، درولینگ، صدای \lr{muffled}: بیانگر عفونت عمقی.
\end{itemize}
آبسهٔ پری‌تونسیلار: مشابه قبلی؛ بیشتر کودکان با سن بالاتر را مبتلا می‌کند.
\FeverOlderUnclear{انتهای سطر آبسهٔ پری‌تونسیلار ممکن است «درگیر می‌کند» خوانده شود.}

\subsection*{عفونت‌های پوست و بافت نرم}
\begin{itemize}
\item سلولیت، آبسه.
\item همراه با تب.
\item در کودکان کم‌سن، باسن ناحیهٔ شایع برای درگیری است.
\end{itemize}

\subsection*{استئومیلیت و آرتریت سپتیک}
تب و عدم وزن‌گذاری.

\subsection*{تظاهر سپسیس}
\lr{ill appearance}، تاکی‌پنه، تاکی‌کاردی.
شوک ممکن است تظاهر اولیهٔ مننگوکوکسمی باشد.

\subsection*{تب بدون یافتهٔ دیگر}
در کودک \lr{2--24 month}، با احتمال \lr{5--10\%}: \lr{UTI}.
\begin{itemize}
\item بیشترین ریسک در دختران و پسران ختنه‌نشده.
\item ختنه‌شده: \FeverOlderMath{<}\lr{\%0.5}.
\end{itemize}
\textbf{یادداشت نماد:} در پایین تصویر، علامت «<» در سمت چپ و «\lr{\%0.5}» در سمت راست آن نوشته شده است؛ ترتیب افقی منبع حفظ شده است.

\FeverOlderPage{۳}{03}
\FeverOlderHeader{۳}
\subsection*{\lr{Occult Bacteremia}}
\begin{itemize}
\item \lr{B/C} مثبت در کودک \lr{well-appearing} بدون منشأ مشخص.
\item با واکسیناسیون پنوموکوک، احتمال آن بسیار کاهش یافته است.
\end{itemize}

\subsection*{رویکرد کلی به تب در کودکان بزرگ‌تر}
\begin{latin}
Step 1) Child's overall appearance\quad and V/S\quad and Hx and Ph/E
\end{latin}
\textbf{\lr{Step 2)}} اگر کودک \lr{V/S} مختل یا ظاهر \lr{ill} را داشت: بررسی و اقدام سریع.

در تب‌های \lr{prolonged} مهم است تشخیص دهیم:
\begin{latin}
Episodic\\
Persistant
\end{latin}
% "Persistant" is the spelling in the handwritten source.
بیمار با تب \lr{prolonged}:
\begin{itemize}
\item ممکن است عفونت‌های خفیف و راجعه باشد.
\item علل التهابی و هماتولوژیک هم مطرح است.
\item \lr{KD} (بیماری کاوازاکی).
\end{itemize}
\begin{latin}
MIS-C\quad Multi-system Inflammatory syndrome in-children
\end{latin}

\subsection*{وجود \lr{B Symptom}}
لوکمی، لنفوم، \lr{TB}.

اریتم و اگزودأ تونسیل + پتشی کام: فارنژیت استرپتوکوکی.

ویز و کراکل منتشر در سمع ریه:
\begin{itemize}
\item برونشیولیت ویروسی حاد.
\item موضعی: پنومونی باکتریال.
\end{itemize}

\FeverOlderPage{۴}{04}
\FeverOlderHeader{۴}
\textbf{حاشیهٔ پایین چپ:} عدد ۵ نیز در گوشهٔ پایین چپ تصویر دیده می‌شود.

\subsection*{ارزیابی آزمایشگاهی}
\begin{itemize}
\item به صورت روتین اندیکاسیون ندارد در کودک \lr{well-appearing} بدون \lr{LS}.
\item کودک \lr{ill} یا \lr{toxic} یا \lr{V/S} مختل: ارزیابی \lr{lab} ضروری است.
\item \lr{B/C}، \lr{U/C}، \lr{LP}: فعلاً \lr{B/C} ضروری.
\end{itemize}
\FeverOlderUnclear{اختصار انتهای سطر نخست به صورت LS خوانده می‌شود.}

مونونوکلئوز: لنفوپنی + لنف آتیپیک.
\FeverOlderUnclear{واژهٔ پیش از «لنف آتیپیک» در منبع «لنفوپنی» خوانده می‌شود و بدون اصلاح حفظ شده است.}

کودکان نقص ایمنی، ایمنی سرکوب‌شده و کودکان با \lr{Cath} داخل عروقی:
باید \lr{ABT} امپریکال را شروع کنیم.

در بیمار \lr{URT} ویروسی یا عفونت \lr{GI}، تست‌های ویروسی به صورت روتین معمولاً اندیکاسیون ندارد.

\subsection*{\lr{PCR} آنفلوآنزا}
صرفاً در ۴۸ ساعت اول علائم، صرفاً در بیماران \lr{high risk}.
\begin{itemize}
\item بیمار \lr{high risk}: ایمیونوساپرسیو.
\item بیماری مزمن قلبی ـ ریوی.
\item \lr{SSD}.
\item سابقهٔ بستری قبلی.
\item سن زیر ۲ سال.
\end{itemize}
\FeverOlderUnclear{واژهٔ نخست فهرست high risk به صورت «ایمیونوساپرسیو» خوانده می‌شود؛ اختصار منبع SSD است.}

چون تصمیم ما برای دادن دارو را تحت تأثیر قرار می‌دهد.

\subsection*{\lr{Rapid strep testing}}
کودکان \FeverOlderMath{3\leq} سال، با شک به فارنژیت استرپتوکوکی.
سن زیر ۳ سال اگر مواجهه در خانه یا یافته در \lr{Ph/E}.
\textbf{یادداشت نماد:} عدد ۳ و علامت مقایسه، با ترتیب افقی «\FeverOlderMath{3\leq}» در منبع نوشته شده‌اند.

\FeverOlderPage{۵}{05}
\FeverOlderHeader{۵}
\subsection*{ادامهٔ ارزیابی آزمایشگاهی}
در دختران و پسران ختنه‌نشده، اگر تب شدید داشته باشند یا ظرف \FeverOlderMath{2\leq} روز تب کاهش نیابد، باید \lr{U/A} انجام دهند.
\textbf{یادداشت بالای سطر:} «۲ـ۶ ماهه» بالای عبارت نخست نوشته شده است.
\FeverOlderUnclear{بازهٔ حاشیه شبیه «۲ـ۶ ماهه» است؛ رقم انتهایی می‌تواند ۴ خوانده شود. علامت زمان به صورت «\FeverOlderMath{2\leq}» در ترتیب افقی تصویر حفظ شده است.}

وجود کراکل لوکالیزه یا کاهش صداهای تنفسی در کودک تب‌دار: پنومونی.
\begin{itemize}
\item \lr{CXR} زمانی که هایپوکسمی یا دیسترس تنفسی شدید وجود داشته باشد، یا درمان سرپایی شکست خورده باشد.
\item در کودکان بستری \lr{CXR} برای ارزیابی \lr{Complication} پنومونی مثل آمپیم.
\end{itemize}

\begin{itemize}
\item ارزیابی آپاندیسیت: سونوگرافی؛ اگر جواب نداد، \lr{CT} یا \lr{MRI}.
\item ارزیابی آبسه: \lr{CT}.
\end{itemize}

\subsection*{منیج کلی بیماران تب‌دار}
\begin{itemize}
\item ضد تب و مایع‌درمانی در صورت نیاز انجام شود.
\item عفونت‌های ویروسی معمولاً فقط نیازمند درمان حمایتی هستند.
\item \lr{Wait and see}: برای کودک با اوتیت میانی حاد.
\item کودک \lr{toxic}، سپسیس، مننژیت باکتریال: درمان سریع با \lr{ABT}.
\item در سایر موارد، \lr{ABT} زمانی که \FeverOlderMath{15\leq WBC} یا \FeverOlderMath{10<ANC} یا \FeverOlderMath{0.5<PCT}.
\end{itemize}
\FeverOlderUnclear{واژهٔ متصل به «سریع با ABT» به صورت «درمان» خوانده می‌شود. اعداد WBC و ANC در منبع فقط ۱۵ و ۱۰ نوشته شده‌اند؛ واحد یا ضریب هزار اضافه نشده است. علامت WBC دوخطی و علامت ANC یک‌خطی خوانده شده‌اند؛ ترتیب افقی نمادها حفظ شده است.}
