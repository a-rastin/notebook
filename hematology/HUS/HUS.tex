\subsection*{\HUSen{HUS}}
\begin{itemize}
\item \HUShl{آنمی همولیتیک میکروآنژیوپاتیک}، \HUShl{ترومبوسایتوپنی}، آسیب حاد کلیه (\HUShl{\HUSen{AKI}}).
\item شایع‌ترین فرم \HUSen{HUS}: همراه با شیگا توکسین تولیدشده توسط \HUSen{STEC}.
  \begin{itemize}
  \item \HUSen{STEC}: \HUSen{Shiga-toxin producing E. coli}.
  \item \HUSen{STEC-HUS}.
  \end{itemize}
\item \HUSen{STEC-HUS}: بیماری اسهالی به صورت پرودرومال.
  \begin{itemize}
  \item \HUShl{\HUSen{Hemorrhagic enterocolitis}}.
  \item \HUSen{Most common}: \HUSen{E. coli O157:H7}.
  \item غذا و آب آلوده.
  \end{itemize}
\item \HUSen{non-STEC-HUS}:
  \begin{itemize}
  \item استرپ پنومونیه، \HUSen{HIV}.
  \item نقص‌های ژنتیکی سیستم کمپلمان، بدخیمی، \HUSen{SLE}، بارداری.
  \item \HUShl{شدیدتر از \HUSen{STEC-HUS}}.
  \item \HUShl{بیشتر به سمت \HUSen{CKD} می‌رود}.
  \end{itemize}
\end{itemize}

\subsection*{\HUSen{TTP}}
\begin{itemize}
\item یک سندرم میکروآنژیوپاتیک مشابه \HUSen{HUS}.
\item \HUShl{بیشتر درگیری \HUSen{CNS}}، ولی \HUSen{AKI} هم ممکن است دیده شود.
\item درگیری \HUSen{CNS} در \HUSen{HUS} هم ممکن است؛ افتراق ممکن است سخت باشد.
\item در \HUSen{TTP}، سطح \HUShl{\HUSen{ADAMTS13}} \HUShl{به شدت پایین است}.
\end{itemize}

\subsection*{علائم \HUSen{STEC-HUS}}
\begin{itemize}
\item بیشتر در کودکان \HUShl{زیر ۵ سال}.
\item انتروکولیت با اسهال خونی.
\item \HUShl{سپس \HUSen{7--10} روز}: ضعف، لتارژی، اولیگوری یا آنوری.
\item در \HUSen{Ph/E}: رنگ‌پریدگی، پتشی، بی‌قراری.
\end{itemize}

\subsection*{ادامهٔ علائم}
\begin{itemize}
\item \HUShl{افت حجم به عنوان تظاهر اولیه شایع است}.
  \begin{itemize}
  \item برخی بیماران ممکن است \HUSen{overload} باشند.
  \end{itemize}
\item هایپرتنشن در بیمار \HUSen{HUS}:
  \begin{itemize}
  \item یا به علت \HUSen{overload} مایع.
  \item یا به علت آسیب کلیه.
  \end{itemize}
\item درگیری \HUSen{CNS}: \HUShl{تشنج} (در ۲۵\% بیماران).
\item سایر درگیری‌ها: پانکراتیت، انفارکت قلبی، پرفوریشن کولون.
\item در \HUSen{non-STEC-HUS} علائم مشابه است ولی سابقهٔ انتروکولیت اخیر وجود ندارد.
\item \HUShl{کومبس منفی است در \HUSen{HUS}}.
\end{itemize}
