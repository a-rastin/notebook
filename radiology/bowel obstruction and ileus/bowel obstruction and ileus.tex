  \item معمولاً به علت التهاب یک ارگان مجاور مثل پانکراتیت
    \begin{itemize}[nosep]

    \item دایورتیکولیت \BOIarrow\ \en{LLQ}
    \end{itemize}
  \item تقریباً همیشه رودهٔ کوچک درگیر است.
  \item \textbf{\en{Sentinel loops}}
  \end{itemize}
\item ایلئوس فانکشنال به‌شدت انسداد مکانیکی نیست و مقداری گاز عبور می‌کند.
\item در ایلئوس فانکشنال لوکالیزه معمولاً هوا به رکتوم و سیگموئید می‌رسد.
\item در \en{CT} تقریباً همیشه علت ایلئوس فانکشنال لوکالیزه قابل پیدا کردن است.
\item \en{Gas pattern}
  \begin{latin}
  \begin{enumerate}[label=\alph*),leftmargin=*,itemsep=0.5em]
  \item Normal $\longrightarrow$ air in rectum or sigmoid (Yes)
    $\longrightarrow$ Air in small bowel (\textbf{Yes, 1--2 loops})
    $\longrightarrow$ Air in large bowel (Rectum and/or sig)
  \item Localized ileus $\longrightarrow$ Yes
    $\longrightarrow$ 2--3 distended loops
    $\longrightarrow$ Rectum and/or sigmoid
  \item Generalized ileus $\longrightarrow$ Yes
    $\longrightarrow$ \textbf{Multiple distended loops}
    $\longrightarrow$ Yes, distended
  \end{enumerate}
  \end{latin}
\end{enumerate}

  \begin{itemize}[nosep]

  \begin{itemize}[nosep]

  \item عموماً به علت ادهیژن
  \item ریسک بالای استرانگولیشن
  \end{itemize}

  \begin{itemize}[itemsep=0.3em]

  \item وقتی دیامتر سکوم به \textbf{\en{12--15 cm}} برسد
    \begin{itemize}[nosep]
    \item خطر پارگی سکوم
    \end{itemize}
  \end{itemize}
