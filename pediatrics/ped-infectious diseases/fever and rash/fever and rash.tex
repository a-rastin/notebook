\begin{center}
{\Large\bfseries ۹۷ ـ تب و راش}\\
\lr{Fever and rash}
\end{center}
\textbf{یادداشت رونویسی:} متن به ترتیب شش برگه آمده است. شاخه‌ها و فلش‌ها به فهرست تبدیل شده‌اند. عبارت‌های پزشکی منبع بدون تصحیح محتوایی حفظ شده‌اند؛ خوانش‌های نامطمئن مشخص شده‌اند. تصاویر کامل منبع در پیوست همین سند قرار دارند.

\FRpage{۱}{01}
\subsection*{۱ ـ سرخک (\lr{Measles}) (روبئولا)}
\begin{itemize}
\item درگیری دستگاه تنفسی فوقانی و لنف‌نودهای ناحیه‌ای.
\item وارد مونوسیت‌ها می‌شود و در ارگان‌های مختلف پخش می‌شود.
\item به شدت واگیر.
\item در ترشحات تنفسی، خون، ادرار.
\end{itemize}
\textbf{مدت زمان قابل انتقال:}
\begin{itemize}
\item از ۱ تا ۲ روز قبل از شروع علائم.
\item از ۵ روز قبل تا ۴ روز بعد از نمایان شدن راش‌ها.
\end{itemize}
شیرخواران تا ۱ سالگی به دلیل آنتی‌بادی مادری مقاوم هستند.

\textbf{چهار فاز بالینی دارد:}
\begin{itemize}
\item انکوبیشن: ۸ تا ۱۲ روز از مواجهه تا شروع علائم؛ ۱۴ روز تا راش.
\item پرودرومال: ۳ روز؛ سرفه، کوریزا، کنژنکتیویت.
\item \lr{Koplik spots}: مقابل مولر تحتانی؛ ۲۴ تا ۴۸ ساعت باقی‌مانده.
\item \lr{Stimson Line}.
\item اگزانتماتوز: تب بالا (۴۰ تا ۴۰٫۵)؛ راش ماکولار از سر شروع می‌شود.
ظرف ۲۴ ساعت به صورت سفالوکودال، با همین الگو محو می‌شود.
\end{itemize}
\FRunclear{عبارت کنار Koplik spots «۲۴ تا ۴۸ ساعت باقی‌مانده» خوانده می‌شود؛ در منبع کلمهٔ «تحتانی» دیده می‌شود. چهار فاز ذکر شده ولی عنوان جداگانه‌ای برای فاز چهارم نوشته نشده است.}
\begin{itemize}
\item شدت بیماری با شدت راش‌ها مرتبط است.
\item \lr{Hack measles}: اگر راش‌ها به صورت پتشی و خونریزی باشند.
\FRunclear{املای انگلیسی منبع «Hack measles» خوانده می‌شود.}
\item لنفادنوپاتی جنرالیزه: شدیدترین درگیری در سرویکال.
\item اوتیت مدیا، پنومونی، اسهال: در شیرخواران شایع‌تر.
\item درگیری کبدی: در بزرگسالان.
\item میوکاردیت.
\end{itemize}

\FRpage{۲}{02}
\subsection*{سرخک ـ ادامه}
\textbf{آزمایشگاه:}
\begin{itemize}
\item غیراختصاصی است و کمک به تشخیص نمی‌کند.
\item لکوپنی کاراکتریستیک است.
\item اگر \lr{LP}: \FRmath{\uparrow}\lr{pro}، \FRmath{\uparrow} لنفوسیت، گلوکوز \lr{N/L}.
\end{itemize}
\begin{latin}
IgM (1--2 days after rash, for 1--2 months)
\end{latin}
\FRunclear{عدد اول مدت ماندگاری IgM شبیه ۱ است؛ شکل دست‌نویس آن شبیه ۹ نیز دیده می‌شود.}
\textbf{درمان:} هیدریشن، ضد خارش، ویتامین \lr{A} با دوز بالا برای ۲ روز.
% The word above the crossed-out treatment term is خارش; use the correction.
\begin{itemize}
\item فرد با نقص ایمنی سلولی مبتلا به سرخک شود: \lr{giant cell (Hecht) pneumonia}؛ کشنده است.
\item انسفالومیلیت: ۲ تا ۸ روز پس از شروع راش‌ها.
\item \lr{SSPE}: ۸ تا ۱۰ سال پس از ابتلا.
\item واکسن \lr{MMR}: ۱۲ تا ۱۵ ماهگی؛ ۴ تا ۶ سالگی.
\item پروفیلاکسی پس از مواجهه: واکسن سرخک ظرف ۷۲ ساعت؛ ایمونوگلوبولین ظرف ۶ روز.
\end{itemize}

\subsection*{۲ ـ روبلا، سرخجه، سرخک آلمانی، سرخک ۳ روزه}
\begin{itemize}
\item انتقال‌پذیری: ۲ روز قبل تا ۵ تا ۷ روز پس از شروع راش‌ها.
\item ابتلای جنین: سندرم روبلای مادرزادی (\lr{CRS})؛ ریسک عوارض بالا؛ در سه‌ماههٔ اول بسیار خطرناک.
\item معمولاً در فصل بهار.
\item آنتی‌بادی مادری ۶ تا ۹ ماه از شیرخوار محافظت می‌کند.
\end{itemize}

\FRpage{۳}{03}
\subsection*{سرخجه ـ ادامه}
\textbf{تظاهرات بالینی:}
\begin{itemize}
\item انکوبیشن: ۱۴ تا ۱۸ روز.
\item پرودرومال: ممکن است خیلی جزئی باشد.
\item اگزانتماتوز: لنفادنوپاتی رتروآریکولار، پوستریور سرویکال، پوستریور اکسیپیتال.
\FRunclear{کنار فلش لنفادنوپاتی یک علامت یا اصلاح کوچک خط‌خورده وجود دارد که خوانا نیست.}
\item راش ماکولوپاپولر، اگزانتماتوز: برای ۳ روز؛ از صورت شروع می‌شود، گسترش به بدن.
\end{itemize}
\begin{latin}
Forchheimer spots \(\longrightarrow\) Rose colored spots on soft palate
\end{latin}
\begin{itemize}
\item فارنژیت، کنژنکتیویت، آنورکسی، تب لو گرید، سردرد.
\item پلی‌آرتریت دست‌ها: خصوصاً در بزرگسالان خانم.
\end{itemize}
\textbf{آزمایشگاه:}
\begin{itemize}
\item \lr{WBC} نرمال یا پایین است؛ در موارد نادر ترومبوسایتوپنی.
\item تأیید تشخیص: \lr{IgM}؛ از ۵ روز پس از شروع علائم.
\end{itemize}
\begin{itemize}
\item درمان حمایتی است.
\item انسفالیت روبلایی عارضهٔ نادر است.
\item واکسن \lr{MMR}: ۱۲ تا ۱۵ ماهگی، ۴ تا ۶ سالگی.
\item پس از واکسن تا ۲۸ روز نباید باردار شوند.
\item پروفیلاکسی پس از مواجهه: واکسن؛ در موارد باردار حتی ایمونوگلوبولین هم نباید بدهیم.
\end{itemize}

\subsection*{۳ ـ روزئولا اینفانتوم (بیماری ششم)}
ناشی از \lr{HHV-6}.
\begin{itemize}
\item آنتی‌بادی مادری تا ۶ ماه از شیرخوار محافظت می‌کند.
\item تا ۲ تا ۳ سالگی همهٔ کودکان مبتلا می‌شوند.
\end{itemize}

\FRpage{۴}{04}
\subsection*{روزئولا ـ ادامه}
\textbf{تظاهرات بالینی:}
\begin{itemize}
\item تب بالای ۴۰ درجه با شروع ناگهانی برای ۳ تا ۵ روز.
\item راش ماکولوپاپولر: ۱ تا ۳ روز؛ در همهٔ بیماران دیده نمی‌شود.
\item علائم دستگاه تنفس فوقانی، احتقان بینی، پردهٔ تیمپان اریتماتو، سرفه.
\item علت \FRmath{1/3} موارد تب و تشنج است.
\end{itemize}
\begin{itemize}
\item در بدن باقی می‌ماند؛ طی پیوند مغز استخوان مجدداً فعال می‌شود.
\item سرکوب \lr{BM}، هپاتیت، راش، انسفالیت.
\item انسفالیت روزئولا: در \lr{LP} پلئوسیتوز ۲۰۰ تا ۳۰۰، \FRmath{\uparrow}\lr{pro}، گلوکوز \lr{N/L}.
\FRunclear{بازهٔ پلئوسیتوز به صورت «۲۰۰ تا ۳۰۰» خوانده می‌شود؛ در دست‌نویس صفرها فشرده‌اند.}
\item درمان: هیدریشن، ضد خارش؛ در نقص ایمنی: گان‌سیکلوویر، فوسکارنت.
\item علت مرگ: انسفالیت، سندرم هموفاگوسیتوز مرتبط با ویروس.
\end{itemize}

\subsection*{۴ ـ اریتم اینفکتیوزوم (بیماری پنجم)}
پاروویروس \lr{B19}.
\begin{itemize}
\item بحران آپلاستیک در بیماران با آنمی همولیتیک: تالاسمی، \lr{HS}، \lr{SC}.
\FRunclear{اختصار آخر به شکل SC نوشته شده است.}
\item در طول بارداری: برای جنین آنمی شدید و هیدروپس فتالیس.
\item در فصل بهار شایع‌تر است.
\item کودکان زیر ۵ سال کمتر درگیر می‌شوند.
\item طول دورهٔ انکوبیشن: ۴ تا ۱۴ روز.
\end{itemize}
\textbf{حاشیهٔ پایین برگه:} یک علامت کوچک شبیه «ب» با خط زیر آن.

\FRpage{۵}{05}
\textbf{سربرگ چاپی اضافی:} \lr{Date:} و \lr{Subject:} (بدون تاریخ).
\subsection*{اریتم اینفکتیوزوم ـ ادامه}
\textbf{تظاهرات بالینی:}
\begin{itemize}
\item علائم غیراختصاصی خفیف: تب، خستگی، سردرد؛ خفیف.
\item راش: ۷ تا ۱۰ روز بعد.
\item فارنژیت، کنژنکتیویت (هر دو خفیف).
\end{itemize}
\textbf{استیج‌های راش:}
\begin{itemize}
\item \lr{slapped cheek rash}: ۱ تا ۴ روز بعد، استیج بعدی.
\item استیج ۳: \lr{lacy, reticulated rash}؛ ۲ تا ۴۰ روز باقی می‌ماند (معمولاً ۱۱ روز).
\item خارش‌دار است؛ دسکواماسیون نمی‌شود.
\item با ورزش و حمام تشدید می‌شود.
\end{itemize}
\begin{itemize}
\item در کودکان مبتلا به آنمی همولیتیک مزمن:
\end{itemize}
\begin{latin}
Transient aplastic crisis\\
For 7--10 days
\end{latin}
\begin{itemize}
\item در بیماران نقص ایمنی می‌تواند ماندگار شود و آنمی شدید ایجاد کند.
\end{itemize}
\textbf{آزمایشگاه:}
\begin{itemize}
\item رتیکولوسیتوپنی برای ۷ تا ۱۰ روز.
\item آنمی خفیف، ترومبوسایتوپنی، لنفوپنی، نوتروپنی.
\item \lr{IgM} اختصاصی تشخیصی است.
\end{itemize}
\begin{itemize}
\item درمان: حمایتی.
\item ویروس \lr{B19} تراتوژن نیست.
\item زمانی که راش‌ها ظاهر شدند دیگر قابل انتقال نیست.
\end{itemize}

\subsection*{۵ ـ آبله‌مرغان (\lr{VZV})}
\begin{itemize}
\item فعال شدن مجدد از گانگلیا ریشهٔ خلفی: \lr{Shingles}.
\item قابلیت انتقال از ۲ روز قبل تا ۷ روز پس از ظاهر شدن راش.
\item بیشترین سن درگیری: ۱۰ تا ۱۴ سالگی.
\item انتقال \lr{Shigles}: تماس مستقیم.
\FRunclear{در سطر آخر انگلیسی به صورت «Shigles» نوشته شده است.}
\end{itemize}

\FRpage{۶}{06}
\subsection*{آبله‌مرغان ـ ادامه}
\begin{itemize}
\item دورهٔ انکوبیشن: ۱۴ تا ۱۶ روز.
\item پرودروم: ۱ روز طول می‌کشد سپس راش‌ها ظاهر می‌شوند؛ تب، خستگی، آنورکسی.
\item پاپول‌های قرمز کوچک که تبدیل به وزیکول می‌شوند.
\item اول درگیری تنه سپس سر و صورت؛ اندام‌ها کمتر درگیر می‌شوند.
\item راش‌ها در فازهای مختلف وجود دارند؛ بسیار خارش‌دار.
\item راش‌ها ممکن است در مخاط‌ها وجود داشته باشند.
\item لنفادنوپاتی جنرالیزه ممکن است رخ دهد.
\item ظرف ۳ تا ۴ روز بهبود می‌یابد.
\end{itemize}
\textbf{تظاهر بالینی زونا:}
\begin{itemize}
\item درد سوزاننده و تندرنس در یک درماتوم؛ تب و خستگی.
\item پاپول ظرف چند روز از شروع علائم؛ برای ۱ تا ۷ روز.
\item لنفادنوپاتی موضعی.
\item اگر درگیری عصب ۷: فلج صورت + وزیکول در گوش = سندرم رمزی هانت.
\item نورالژی پست هرپتیک: دردی که بیشتر از ۱ ماه طول بکشد؛ در کودکان ناشایع است.
\end{itemize}
\textbf{آزمایشگاه:} غیراختصاصی است.

\textbf{درمان:}
\begin{itemize}
\item ضد خارش، غیر آسپرین، حمام سرد.
\FRunclear{عبارت پس از «ضد خارش» به صورت «غیر آسپرین» خوانده می‌شود؛ به دلیل اتصال حروف، «ضد آسپرین» نیز خوانش محتمل است.}
\item در کودکان عادی آسیکلوویر تجویز نمی‌کنیم.
\item در افراد نقص ایمنی: ظرف ۲۴ ساعت از شروع راش‌ها.
\item افراد بالای ۱۲ سال.
\item \lr{VZIG}: ظرف ۹۶ ساعت پس از مواجهه.
\end{itemize}
