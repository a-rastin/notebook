\subsection*{اتیولوژی}
\begin{itemize}
\item \phhl{ویروس‌ها} شایع‌ترین عامل فارنژیت هستند.
\item \phhl{استرپتوکوک گروه \en{A}} مهم‌ترین باکتری عامل فارنژیت است: استرپ پیوژنز.
\item \en{GAS}: فارنژیت، تونسیلیت، تب اسکارلت.
\item مونونوکلئوز: \en{EBV}، \en{CMV}.
\item کلامیدوفیلا نومونیا: درگیر دستگاه تنفسی تحتانی می‌شود ولی می‌تواند گلودرد هم ایجاد کند.
\item مایکوپلاسما نومونیه: \phhl{پنومونی آتیپیک} ایجاد می‌کند؛ فارنژیت خفیف.
\item بسیاری از ویروس‌ها می‌توانند فارنژیت بدهند ولی در \phhl{آدنوویروس‌ها} فارنژیت تظاهر اصلی است.
\end{itemize}
\subsection*{اپیدمیولوژی}
\begin{itemize}
\item گلودرد علامت اولیه در \en{1/3} بیماری‌های دستگاه تنفسی فوقانی است.
\end{itemize}

\phpage{۲}{pic02.jpg}{20261010\_191502207.jpg}
\begin{itemize}
\item فارنژیت استرپتوکوکی \phhl{در سن زیر \en{2--3} سال بسیار ناشایع} است.
\item در \phhl{سنین مدرسه} شایع می‌شود سپس دوباره کاهش می‌یابد.
\item فارنژیت استرپتوکوکی به سرعت به هم‌کلاسی‌ها، خواهر و برادر منتقل می‌شود.
\item عفونت‌های ویروسی عمدتاً از طریق \en{Close Contact} منتقل می‌شوند.
\item مونونوکلئوز ناشی از \en{EBV}، \en{CMV}: فارنژیت ممکن است تنها تظاهر اصلی بیماری باشد.
\item در بزرگسالان، \en{young adults}، شایع‌تر است.
\end{itemize}
\subsection*{تظاهر بالینی}
\begin{itemize}
\item التهاب فارنکس: سرفه، گلودرد، دیسفاژی، تب.
\item اگر درگیری تانسیل‌ها بارز باشد: تانسیلوفارنژیت.
\item فارنژیت استرپتوکوکی:
  \begin{itemize}
  \item \phhl{\en{Rapid onset}}.
  \item گلودرد بارز (\en{prominent}).
  \item \phhl{تب متوسط تا شدید}.
  \item سردرد، تهوع، استفراغ، درد شکمی: شایع است.
  \end{itemize}
\end{itemize}

\subsection*{خط اول درمان}
\phhl{پنی‌سیلین یا آموکسی‌سیلین}.
\begin{latin}
\begin{itemize}
\item Oral penicillin V: 250--500 mg/dose, BD, TDS $\times$ 10 days.
\item Amp penicillin G, IM, Single dose:
  \begin{itemize}
  \item For $\leq 27$ kg: 600,000 U.
  \item Larger children: 1.2 million U.
  \end{itemize}
\item Oral \phhl{Amoxicillin}: 50 mg/kg; Max: 1200 mg, OD $\times$ 10 days.
\end{itemize}

\noindent Allergy to penicillin
\begin{itemize}
\item Amp Cephalexin: 25 mg/kg/dose, BD; MAX: 500 mg/dose $\times$ 10 days.
\item Amp Clindamycin: 6--7 mg/kg/dose, TDS; MAX: 300 mg/dose $\times$ 10 days.
\end{itemize}

\noindent Allergy to $\beta$-lactams
\begin{itemize}
\item Azithromycin: 12 mg/kg (Max: 500 mg/dose) on day 1, followed by
      6 mg/kg/dose (Max: 250 mg), OD on day 2--5.
\end{itemize}
\end{latin}
