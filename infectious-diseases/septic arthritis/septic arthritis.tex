\subsection*{ارگانیسم‌ها}
\begin{itemize}

  \begin{itemize}
  \item پس از \textbf{عمل جراحی یا تروما نافذ}: استاف اورئوس.
  \item سالمندان، خصوصاً آن‌هایی که بیماری زمینه‌ای دارند: باسیل گرم منفی، پنوموکوک، استرپ \en{B}، هموفیلوس.
  \item آرتروسکوپی، جراحی تعویض مفصل: استاف کوآگولاز منفی.
  \item نفوذ جسم تیز به داخل کفش پا: سودومونا آئروژینوزا.
  \item کمبود اولیهٔ ایمونوگلوبولین: مایکوپلاسما.
    \begin{itemize}
    \item آرتریت مایکوپلاسما: \textbf{تجویز سریع تتراسایکلین، \en{IVIG}}.
    \item در غیر این صورت \textbf{تخریب دائمی} مفصل و غضروف.
    \end{itemize}
  \item \en{DM}، آرتروز، همودیالیز، بدخیمی: استاف اورئوس، باسیل گرم منفی.
  \item درمان \en{RA} با مهارکنندهٔ \en{TNF} (\textbf{آباتاسپت، اینفلیکسیمب}): مایکوباکتریایی.
  \item الکلی‌ها، اختلالات ایمنی هومورال، هموگلوبینوپاتی‌ها: \textbf{پنوموکوک}.
  \item \en{IVDU}: استاف، استرپ، سودومونا، گرم منفی‌ها.
  \end{itemize}
\end{itemize}

\subsection*{بررسی‌ها}
\begin{itemize}
\item \textbf{قبل از شروع \en{AB}: نمونهٔ خون، مفصل، مایع سینوویال}.
\item در آنالیز مایع مفصلی: \textbf{\en{LDH} بالا، گلوکز پایین}.
\item مایع مفصلی باید حتماً از نظر \textbf{کریستال} بررسی شود: \en{gout}، \en{pseudogout}.
\item \en{ESR} و \en{CRP} بسیار بالا است. پروکلسیتونین: حساسیت ۸۰\%.
\item اگر کشت منفی شد ولی شک عفونی بالا بود: اسپکترومتری به روش \textbf{\en{MALDI-TOF}}.

  \begin{itemize}

  \item کاهش فضای مفصل، اروژن استخوان: پس از آن می‌آید.
  \end{itemize}
\item \textbf{سونو}: برای بررسی افیوژن \textbf{هیپ}.
\item \textbf{\en{MRI}، \en{CT}}: برای مفاصل \textbf{ساکروایلیاک، استرنوکلاویکولار، ستون فقرات}.
\end{itemize}

\subsection*{ادامهٔ درمان اختصاصی}
\begin{itemize}
\item گرم منفی روده‌ای: ۳--۴ هفته / سفالوسپورین نسل ۲ یا ۳ \en{IV}، یا فلوروکینولون مثل لوو \en{IV} یا \en{PO}.
\item سودومونا آئروژینوزا:
  \begin{itemize}
  \item حداقل ۲ هفته / آمینوگلیکوزید + پنی‌سیلین وسیع‌الطیف (پیپراسیلین).
  \item یا سفتازیدیم (سفالوسپورین ضد سودومونا).
  \item یا فلورو مثل سیپرو + پنی‌سیلین یا سفالوسپورین.
  \item اگر بعد از ۲ هفته خوب نشد: ۲ هفته تمدید درمان.
  \end{itemize}
\item اساس درمان آرتریت باکتریایی: تجویز سریع \en{AB} سیستمیک + \textbf{درناژ چرک} (اگر لازم شد مکرر).
\item \textbf{آرتروتومی} در \textbf{آرتریت هیپ} خیلی خوب جواب می‌دهد.
\item تا زمان رفع التهاب، وزن خود را بر روی مفصل \textbf{قرار ندهد}.
\item \textbf{حرکات مکرر پاسیو} در مفصل درگیر انجام شود.
\item بی‌حرکت نمودن مفصل لازم نیست.
\end{itemize}

  \begin{itemize}

\item آسپیراسیون مایع مفصلی در \en{DGI} منفی است و ۱۰٬۰۰۰ تا ۲۰٬۰۰۰ سلول دارد.
\end{itemize}

\subsection*{آرتریت سپتیک گنوکوکی}
\begin{itemize}
\item \textbf{همواره متعاقب \en{DGI}} رخ می‌دهد.
\item درگیری \textbf{مونوآرتیکولار}: هیپ، زانو، مچ پا، مچ دست.
\item مایع سینوویال: \textbf{\en{$\mathrm{WBC}>50{,}000$}}.
\item \textbf{کشت خون همواره منفی}، کشت مایع مفصلی در ۴۰\% مثبت.
  \begin{itemize}
  \item محیط کشت مناسب برای گنوکوک: \textbf{آگار--تایر--مارتین}.
  \end{itemize}
\item تشخیص \en{DNA} گنوکوک در مایع سینوویال بر اساس \textbf{\en{NAA}}: بسیار حساس.
\item زنان در هنگام قاعدگی یا \textbf{حاملگی} بیشترین ریسک ابتلا به \en{DGI} و آرتریت گنوکوکی.
\item درمان اولیه: \textbf{سفتریاکسون} / ۷ روز درمان.
\item بیمار \en{DGI} باید درمان کلامیدیا تراکوماتیس را نیز دریافت کند مگر اینکه \en{R/O} شده باشد: \textbf{آزیترومایسین}.
\item شریک جنسی هم آزمایش و درمان باید شود.
\end{itemize}

\subsection*{درمان عفونت در مفاصل مصنوعی}
\begin{itemize}
\item اغلب نیاز به \textbf{تعویض پروتز}.
\item \textbf{\en{AB} وریدی ۴--۶ هفته پیش از جراحی} تعویض مفصل باید داده شود.
\item چه زمانی نیاز به تعویض نیست؟
  \begin{itemize}
  \item عفونت استرپ یا پنوموکوک در صورتی که شل‌شدگی پروتز در گرافی دیده نشود.
  \item آرتروتومی یا آرتروسکوپی.
  \end{itemize}
\item \en{AB} در استاف: ریفامپین خوراکی + کینولون یا پنی‌سیلین ضد استاف یا ونکو.
\item \en{AB} پروفیلاکسی برای اقدامات دندان‌پزشکی و اورولوژی در بیمار با تعویض مفصل اندیکاسیون ندارد.
\item سنگ‌شکنی و جراحی روده: پروفیلاکسی با \textbf{سفازولین}؛ برای \en{MSSA}.
\end{itemize}
