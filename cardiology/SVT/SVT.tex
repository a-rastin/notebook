\svtpage{۱}{\en{65 -- SVT}}
\subsection*{تاکی‌کاردی فوق‌بطنی (\en{SVT})}
\subsection*{تعریف}
یک ترم کلی است. همهٔ تاکی‌کاردی‌هایی که:
\begin{itemize}
\item ریت دهلیز و/یا بطن بالای ۱۰۰ باشد.
\item در مکانیسم آن، باندل هیس یا بالاتر درگیر باشد.
\end{itemize}

\subsection*{انواع \en{SVT}}
\begin{itemize}
\item \en{Sinus Tachy}
\item \en{Atrial Tachy}: \en{FAT, MAT}
\item ماکرو ری‌انترنت \en{AT}: فلاتر
\item \en{JT}
\item \en{AVNRT}
\item \en{AVRT}
\end{itemize}

\subsection*{\en{PSVT}}
یک سندرم بالینی است. یک ریتم رگولار با تاکی‌کاردی با شروع و ختم ناگهانی.

\subsection*{اپروچ}
\begin{itemize}
\item بیمار با تپش قلب.
\item شرح حال: سبکی سر، تنگی نفس، احساس بد در گلو، درد قفسهٔ سینه.
\item سینوس تاکی‌کاردی علامت‌دار: در حدود ۱۳۰ تا ۲۲۰ است.
\item \en{IAST}: علامت‌ها ناگهانی شروع می‌شوند ولی ممکن است تدریجی هم باشند.
  \begin{itemize}
  \item پایان همواره تدریجی است.
  \item با تنگی نفس، سبکی سر، خستگی همراه است.
  \end{itemize}
\end{itemize}

\svtpage{۲}{\en{65 -- SVT}}
\begin{itemize}
\item \en{SVT}: شروع اغلب ناگهانی، ریت بسیار بالا.
  \begin{itemize}
  \item شروع به دنبال یک حرکت خاص.
  \item دویدن ناگهانی، خم شدن.
  \item اپیزودها چند دقیقه تا چند ساعت.
  \item پایان ناگهانی.
  \end{itemize}
\item معمولاً به مانور واگ جواب می‌دهد.
\item معمولاً در شروع با سبکی سری همراه هستند که پره‌سنکوپ است. با این حال سنکوپ نامعمول است.
\item اغلب با اضطراب همراه هستند.
\item پلی‌اوری حین و بعد از \en{SVT} شایع است.
\end{itemize}

\subsection*{معاینهٔ فیزیکی}
بررسی از نظر اختلالات دریچه‌ای، نارسایی قلبی، تیروتوکسیکوز.

\subsection*{\en{EKG}}
باید یک \en{EKG} دوازده‌لیدی بگیریم، در زمان ریتم سینوسی.
\begin{itemize}
\item بررسی از نظر ابنورمالیتی \en{P wave} و بلوک‌ها.
\end{itemize}

\subsection*{آزمایش‌ها}
بیوشیمی، \en{TFT}، تروپونین (در اورژانس).

\subsection*{اکوکاردیوگرافی}
\begin{itemize}
\item اختلالات دریچه‌ای.
\item سایز حفره‌ها.
\end{itemize}

\svtpage{۳}{\en{65 -- SVT} ـ براونوالد}
باید آریتمی بیمار را داکیومنت کنیم.
\begin{itemize}
\item اگر هر روز تکرار می‌شود: \en{Single 24 h Holter}.
\end{itemize}

\begin{itemize}
\item \en{IAST}: ریت تدریجاً ظرف چند ثانیه تا چند دقیقه بالا می‌رود.
\item \en{FAT}: ناگهانی و در لحظه به حداکثر ریت می‌رسد.
\end{itemize}

برای بیماران با علائم کلاسیک شروع و ختم ناگهانی یک تاکی‌کاردی که قویاً به نفع \en{SVT} باشد، داکیومنت کردن \en{SVT} ضروری نیست و می‌توان از همان ابتدا \en{EPS} انجام داد.

\svtpage{۱}{۲۴۶ ـ اپروچ به \en{SVT} ـ هاریسون}
\subsection*{\en{Pathologic SVT}}

\subsubsection*{\en{A: Tachy Originating from Atrium}}
\begin{itemize}
\item \en{IAST}: سینوس تاکی‌کاردی بدون علت زمینه‌ای.
\item \en{FAT}: از نقطه‌ای غیر \en{SA} منشأ می‌گیرد.
\item \en{Atrial Flutter}: ریت بالای ۳۰۰ است.
\item \en{AF}
\item \en{MAT}: معمولاً بیماران ریوی حین تشدید بیماری.
\end{itemize}
