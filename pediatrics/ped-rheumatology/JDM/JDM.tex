\sourcehead{۱}{Juvenile Dermatomyositis -- 91}
\begin{itemize}
\item اتیولوژی \en{JDM} ناشناخته است.
\item \en{JDM}: فعال شدن لنفوسیت \en{T} و \en{B}؛ واسکولوپاتی عروق کوچک عضلات اسکلتی، ناشی از رسوب کمپلکس ایمنی؛ التهاب عروق، عضلات، پوست.
\item \en{JDM}، تریگرها: \hl{عفونت}، \hl{واکنش آلرژیک}، \hl{نور خورشید}.
\item \en{JDM}: بیماری نادر؛ در \en{F} اندکی از \en{M} شایع‌تر است.
\item تمام سنین را می‌تواند درگیر کند؛ پیک سن \en{5--14} سال.
\end{itemize}

\subsection*{تظاهرات بالینی}
\begin{itemize}
\item \hl{شروع ناگهانی خستگی}، ضعف و ضعف عضلانی پیش‌رونده؛ \hl{تب \en{lowgrade}}؛ \hl{راش}.
\end{itemize}

\subsection*{الگوی درگیری عضله}
\begin{itemize}
\item بیشتر عضلات پروگزیمال: \hl{کمربند شانه‌ای و لگنی}؛ عضلات شکم و گردن.
\item کودک نمی‌تواند از روی زمین یا صندلی بلند شود یا از پله بالا برود: \hl{\en{Positive Gower sign}}.
\item در موارد شدید، بیمار نمی‌تواند بنشیند.
\item درگیری عضلات حنجره: صدای \en{nasal}.
\item درگیری عضلات حلق: بلع دشوار.
\end{itemize}

\subsection*{الگوی راش}
\begin{itemize}
\item صورت، گونه‌ها؛ ممکن است شانه‌ها و کمر را نیز درگیر کند: \hl{\en{shawl sign}}.
\item تغییر رنگ پلک‌ها: \hl{\en{heliotrope discoloration}}.
\item پلاک‌های قرمز به بنفش روی \en{Knuckles} و سطوح اکستانسور مفاصل: \hl{\en{Gottron papules}}.
\end{itemize}

\sourcehead{۲}{JDM -- 91}
\begin{itemize}
\item اریتم \en{periungual} و \en{dilated nail fold capillaries}.
\item علائم ناشایع‌تر در \en{JDM}: واسکولیت پوستی با التهاب، اریتم و تخریب پوست.
\item درگیری مفاصل: در ۱۵\% بیماران آرتریت رخ می‌دهد؛ بیشتر \hl{مفاصل کوچک}، ولی می‌تواند هر مفصلی را درگیر کند.
\item در \en{JDM}: \hl{لنفادنوپاتی احتمال دارد}؛ \hl{پدیدهٔ رینود احتمال دارد}.
\end{itemize}

\subsection*{بررسی‌ها}
\begin{itemize}
\item[$\ast$] اکثر بیماران \en{JDM} شواهد \hl{آزمایشگاهی التهاب را ندارند}.
\item \en{JDM}: سطح خونی آنزیم‌های موجود در عضلات بالا می‌رود: \hl{\en{ALT, AST, CPK, LDH, Aldolase}}.
\item \en{EMG} و بیوپسی کمک‌کننده هستند.
\item \en{MRI} التهاب عضلات را نشان می‌دهد.
\item[$\ast$] \textbf{درصد بسیار کمی از بیماران راش ندارند.}
\end{itemize}

\subsection*{درمان \en{JDM}}
\begin{itemize}
\item متوترکسات + دورهٔ ۶ ماههٔ کورتیکواستروئید.
\item \en{IVIG} کمک‌کننده است.
\item موارد شدید و راجعه: سیکلوسپورین، مایکوفنولات، آزاتیوپرین، تاکرولیموس، سیکلوفسفاماید.
\item \hl{هیدروکسی‌کلروکین} برای \hl{راش‌های پوستی}.
\item نور خورشید می‌تواند باعث شعله‌ور شدن بیماری شود.
\item مکمل \hl{\en{Ca}} و \hl{\en{vitD}} فعال باید دریافت کنند.
\end{itemize}

\subsection*{کرایتریا تشخیص \en{JDM}}
باید \hl{۴ مورد از ۵ مورد} را داشته باشند:
\begin{enumerate}
\item راش تیپیکال.
\item ضعف سیمتریک عضلات پروگزیمال.
\item افزایش سطح خونی آنزیم‌های عضلانی.
\item \en{EMG} تیپیکال \en{JDM}.
\item بیوپسی مثبت برای \en{JDM}.
\end{enumerate}

\subsection*{عوارض}
\begin{itemize}
\item مهم‌ترین عارضهٔ \en{JDM}: \hl{کلسینوز} است.
\item کلسیفیکاسیون دیستروفیک در پوست و بافت نرم بدن می‌تواند رخ دهد:
\en{mild calcinosis to calcinosis universalis}.
\item ریسک فاکتورها برای کلسینوز: واسکولیت پوستی، بیماری طول‌کشیده، تأخیر در شروع درمان.
\item بیمار \en{JDM} که دچار واسکولیت می‌شود: ریسک پرفوریشن \en{GI} و \en{GIB}.
\item عوارض دیگر: لیپوآتروفی؛ مقاومت به انسولین: \hl{\en{DM II}}.
\end{itemize}
