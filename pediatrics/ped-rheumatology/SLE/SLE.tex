\pagehead{۱}
\begin{itemize}
\item طی \en{SLE} مقدار زیادی \en{Ab} به خودش تولید می‌شود؛ اختصاصی و غیراختصاصی.
\item \en{Abs} کمپلکس‌های ایمنی ایجاد می‌کنند که در ساختارهای میکروواسکولار گیر می‌افتند؛ التهاب + ایسکمی.
\end{itemize}

\subsection*{اپیدمیولوژی \en{SLE}}
\begin{itemize}
\item \hl{زنان \en{20--40} سال}.
\item ۵\% موارد \en{SLE} در کودکی ظاهر می‌شوند؛ اکثراً نزدیک بلوغ.
\item در کودکان توزیع جنسی برابر است.
\end{itemize}

\subsection*{تظاهرات بالینی \en{SLE}}
\begin{itemize}
\item الگوی بیماری: شروع ناگهانی \en{(fulminant, abrupt)} یا شروع تدریجی \en{(Indolent)}.
\item علائم غیراختصاصی: \hl{ضعف و خستگی شدید}، \hl{تب \en{low grade}}، \hl{کاهش وزن}.
\item \hl{درگیری پوستی در ۹۰\% بیماران} \en{SLE} دیده می‌شود.
\item راش برجستهٔ اریتماتو؛ به آن \hl{\en{Malar butterfly rash}} می‌گویند.
\item \hl{پل بینی را درگیر می‌کند}، پیشانی؛ چین نازولبیال را درگیر نمی‌کند.
\item فتوسنسیتیویتی: در آفتاب.
\item مالار راش و حساسیت به نور با درمان کنترل می‌شوند.
\item دیسکوئید لوپوس: فرایند التهابی؛ درمو اپی‌درمال جانکشن را درگیر می‌کند.
\item \hl{اسکارهای دائمی} به جا می‌گذارد؛ \en{loss of pigmentation}؛ در اسکالپ: آلوپسی دائمی.
\end{itemize}

\pagehead{۲}
\begin{itemize}
\item آلوپسی، \hl{رینود} و لیویدورتیکولاریس در \en{SLE} دیده می‌شوند ولی اختصاصی نیستند.
\item زخم در حفرهٔ دهان و بینی در \en{SLE} دیده می‌شود؛ حتی پرفوریشن سپتوم بینی.
\item \hl{لنفادنوپاتی و اسپلنومگالی} در \en{SLE} شایع است (به دلیل تحریک سیستم رتیکولواندوتلیال)؛ \hl{لنفادنوپاتی آگزیلاری} برای \en{SLE} حساس است.
\item سروزیت در \en{SLE}: \en{chest pain}، پلورال و یا پریکاردیال فریکشن راب؛ افیوژن و مایع.
\item \hl{درگیری کلیوی در \en{SLE} کودکان} شایع است؛ از پروتئین‌اوری و هماچوری میکروسکوپیک تا سندرم نفروتیک و نارسایی کلیه. فشار خون و ادم می‌تواند دیده شود.
\item آرتریت و آرترالژی در \en{SLE} شایع است؛ \hl{معمولاً دفورمیتی ایجاد نمی‌کند}. معمولاً \hl{مفاصل کوچک دست} را درگیر می‌کند؛ هر مفصلی ممکن است.
\item میالژی، میوزیت، ضعف عضلانی، خستگی، تندرس عضلانی؛ ممکن است دیده شود.
\item درگیری \en{CNS}: از افت عملکرد تحصیلی و عدم تمرکز تا تشنج، سایکوز و استروک.
\end{itemize}

\subsection*{یافته‌های هماتولوژیک در \en{SLE}}
\begin{itemize}
\item \hl{کومبس مثبت}؛ آنمی همولیتیک خودایمن + رتیکولوسیتوز.
\item لکوپنی: \en{$\mathrm{WBC}<4000$} در دو نوبت.
\item لنفوپنی: \en{$\mathrm{Lym}<1500$} در دو نوبت.
\item \hl{ترومبوسایتوپنی}: \en{$\mathrm{Plt}<100{,}000$}.
\item آنمی بیماری مزمن.
\end{itemize}

\pagehead{۳}
\begin{itemize}
\item \en{VDRL} مثبت کاذب برای بیش از ۶ ماه در \en{SLE} می‌تواند دیده شود.
\item \hl{\en{ANA} مثبت}: اختصاصی نیست ولی در ۹۹\% بیماران \en{SLE} دیده می‌شود؛ معمولاً در \hl{تیترهای خیلی بالا}.
\item \hl{اگر \en{ANA} منفی شد، عملاً \en{SLE} رد می‌شود.}
\item \en{anti-ds DNA Ab}: شک می‌کنیم به \en{SLE}؛ برای \en{SLE} اختصاصی است. مقدار آن برای \hl{بررسی فعالیت بیماری} اندازه‌گیری می‌شود.
\item \en{anti Sm Ab}: اختصاصی برای \en{SLE}، ولی فقط در ۳۰\% بیماران دیده می‌شود.
\item \en{anti Ro (SSA), anti La (SSB)}: در \en{SLE} و سندرم شوگرن دیده می‌شود.
\item آنتی‌فسفولیپید آنتی‌بادی: هم در \en{SLE}، هم در سایر بیماری‌های روماتولوژیک. \hl{ریسک ترومبوز شریانی و وریدی} را بالا می‌برد؛ آنتی‌کاردیولیپین آنتی‌بادی، \en{VDRL} مثبت کاذب، \en{PTT} طولانی.
\item افزایش سطح آنتی‌بادی‌ها در گردش: مصرف پروتئین‌های کمپلمان؛ \hl{سطح پایین \en{$C_3$} و \en{$C_4$}}. عملکرد کمپلمان کاهش می‌یابد: \hl{\en{\dn CH50}}.
\item \hl{از سطح کمپلمان برای بررسی درمان استفاده می‌شود.}
\item بررسی عملکرد کلیه: \en{BUN} سرم، \en{Cr} سرم؛ هماچوری، پروتئین‌اوری.
\item اگر \hl{میوزیت} وجود داشته باشد، \hl{آنزیم‌های عضلانی \en{\up}} می‌روند؛ آلدولاز.
\end{itemize}

\subsection*{بررسی درگیری \en{CNS}}
\begin{itemize}
\item افزایش پروتئین \en{CSF}.
\item \hl{افزایش \en{IgG Index}}: نسبت \en{$\dfrac{\mathrm{IgG}}{\mathrm{Alb}}$} مایع \en{CSF} به سرم.
\item الگوی اختصاصی درگیری \en{CNS} در \hl{\en{MRI} با تزریق گادولینیوم}.
\end{itemize}

\subsection*{تولید مثل در \en{SLE}}
\begin{itemize}
\item سقط‌های خودبه‌خودی مکرر.
\item لوپوس نوزادی.
\item \hl{بلوک قلبی جنین}.
\end{itemize}

\subsection*{درمان \en{SLE}}
\begin{itemize}
\item کورتیکواستروئید نقش اصلی درمان است: پالس متیل‌پردنیزولون؛ پردنیزولون خوراکی با دوز بالا (تا \hl{\en{2 mg/kg}}).
\item \en{NSAID} برای آرتریت و آرترالژی.
\item هیدروکسی‌کلروکین: درگیری‌های پوستی؛ درمان نگهدارنده. مزایا: رمیشن طولانی‌تر، عوارض عروقی کمتر، بهبود دیس‌لیپیدمی.
\item \hl{سیکلوفسفاماید}: \hl{نفریت لوپوسی}، درگیری \en{CNS}.
\item در بیمارانی که استروئید را تحمل نمی‌کنند یا نمی‌خواهیم استروئید بدهیم: \en{Steroid-sparing agents}:
\end{itemize}
\begin{latin}
\begin{enumerate}
\item Azathioprine
\item Methotrexate
\item Mycophenolate-Mofetil
\end{enumerate}
\end{latin}
\begin{itemize}
\item کلسیم و \en{vit D} باید بدهیم.
\item \hl{درمان هایپرلیپیدمی اندیکاسیون دارد.}
\end{itemize}
