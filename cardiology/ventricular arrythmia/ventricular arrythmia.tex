\varrpage{۱}{براونوالد ـ ۹۷ ـ آریتمی‌های بطنی}
\subsection*{آریتمی‌های بطنی}
\begin{itemize}
\item \en{PVC}
\item \en{VT}: \en{Sustained, Nonsustained}
\item \en{VF}
\end{itemize}
آریتمی‌های بطنی می‌توانند در همهٔ بیماری‌هایی که بطن‌ها را درگیر می‌کنند ایجاد شوند و ممکن است تظاهر اولیهٔ آن بیماری باشند.

آریتمی‌های بطنی در افرادی که اختلال ساختاری قلبی ندارند و ابنورمالیتی کانال‌های یونی نیز ندارند معمولاً خوش‌خیم هستند: ایدیوپاتیک.

\subsection*{\en{PVC}}
\begin{itemize}
\item هم پره‌مچور است، هم \en{QRS} غیرطبیعی است و هم معمولاً \en{wide} است.
\item موج \en{P} ندارد.
\item موج \en{T} آن معمولاً پهن است و خلاف جهت \en{QRS}.
\item جبران کامل دارد: فاصلهٔ \en{RR} کمپلکس طبیعی قبل و بعد \en{PVC} دقیقاً ۲ برابر \en{RR} عادی است.
\item اگر موج رتروگراد داشته باشد، معمولاً \en{P} آن رتروگراد و در آخر کمپلکس \en{QRS} دیده می‌شود.
  \begin{itemize}
  \item در این صورت ممکن است یک \en{Incomplete Compensatory pause} رخ دهد.
  \end{itemize}
\end{itemize}

\varrpage{۲}{براونوالد ـ ۹۷ ـ آریتمی بطنی}
\begin{itemize}
\item ممکن است \en{PVC} بین دو بیت سینوسی بیاید و اصلاً \en{pause} ایجاد نکند: \en{Interpolated PVC}.
\end{itemize}
\subsection*{محل \en{PVC}}
\begin{itemize}
\item اگر موج \en{S} دامیننت در \en{$V_1$}: \en{LBBB like}؛ معمولاً در بطن راست یا سپتوم بین‌بطنی.
\item اگر موج \en{R} دامیننت در \en{$V_1$}: \en{RBBB like}؛ معمولاً در بطن چپ.
\end{itemize}
\subsection*{\en{PVC}}
\begin{itemize}
\item \en{Unifocal}: شکل \en{QRS} در \en{PVC}ها یکسان است.
\item \en{Multifocal}:
  \begin{itemize}
  \item بیشتر با \textbf{بیماری‌های ساختاری قلب} مرتبط است.
  \end{itemize}
\item \en{Bigeminy}: به ازای هر \en{PVC} یک بیت سینوسی.
\item \en{Trigeminy}: به ازای هر \en{PVC} دو بیت سینوسی.
\item فاصله بین \en{PVC} و سینوسی می‌تواند ثابت یا متغیر باشد.
\item به ۲ \en{PVC} متوالی بدون بیت سینوسی، \en{PVC Couplet} می‌گویند.
\item به ۳: \en{PVC Triplet}.
\end{itemize}
\subsection*{تعریف \en{Nonsustained VT}}
۳ \en{PVC} متوالی در ۳۰ ثانیه.

\varrpage{۳}{\underline{\hspace{5cm}}}
\begin{itemize}
\item اگر \en{VT}، پلی‌مورفیک، \en{$\mathrm{rate}>220$}، شروع نزدیک قلهٔ موج \en{T} بیت سینوسی:
  \begin{itemize}
  \item خطر بالای سنکوپ و مرگ.
  \end{itemize}
\end{itemize}
\subsection*{علائم \en{PVC}}
\begin{itemize}
\item بی‌علامتی.
\item تپش قلب، سبکی سر، ضعف، خستگی.
\end{itemize}
حجم ضربه‌ای خود \en{PVC} پایین است و نبض ایجاد نمی‌کند.

\subsection*{برخورد به \en{PVC}}
\begin{itemize}
\item باید ببینیم آیا \textbf{بیماری ساختاری یا الکتریکی قلبی} وجود دارد یا نه.
\item \en{EKG} می‌گیریم: هر اختلال در \en{EKG} غیر از این‌ها باید بررسی بیشتر شود.
  \begin{itemize}
  \item بررسی کاردیومیوپاتی و \en{CAD}.
  \end{itemize}
\end{itemize}
\en{PVC}های \textbf{بدون علامت} نیاز به درمان ندارند تا زمانی که آن‌قدر شدت آن‌ها بالا برود که \textbf{اختلال عملکرد بطنی} ایجاد کند.

کاهش کافئین توصیه می‌شود: علائم \en{PVC} \en{$\downarrow$}.

اگر نیاز به درمان بود: \textbf{بتابلاکر نخستین}؛ \en{PVC} را کم می‌کند ولی علائم بهتر می‌شوند.

\varrpage{۴}{\underline{\hspace{5cm}}}
\begin{itemize}
\item \en{L-type CCB} (غیردی‌هیدروپیریدین): وراپامیل، دیلتیازم؛ کاهش مؤثر.
\item آنتی‌آریتمی‌های کلاس \en{IC} نیز کاهش مؤثر هستند:
  \begin{itemize}
  \item \en{Flecainide}، پروپافنون.
  \end{itemize}
\item آمیودارون نیز کاهش مؤثر است ولی به دلیل سمیت خودش آن \textbf{استفاده نمی‌شود}.
\item \en{Catheter Ablation}: زمانی که درمان ضروری است و بتابلاکرها مؤثر نبوده‌اند.
  \begin{itemize}
  \item خصوصاً اگر \en{PVC} تک‌کانونی باشد.
  \end{itemize}
\end{itemize}
\subsection*{\en{PVC} در بیمار با اختلال و بیماری ساختاری قلب}
\begin{itemize}
\item اگر اختلال عملکرد بطن نداشته باشیم، درمان دادن \en{PVC} کمکی به بهبود پروگنوز نمی‌کند.
\item \textbf{\en{PVC} علامت‌دار با اختلال عملکرد بطنی}: درمان:
  \begin{itemize}
  \item آمیودارون.
  \item \en{Cath Ab}.
  \end{itemize}
\item \textbf{طی \en{MI}، \en{PVC} را درمان نمی‌کنیم.}
\item \en{PVC} که \en{VF} مکرر ایجاد کند: \en{Quinidine}.
\end{itemize}

\varrpage{۵}{\underline{\hspace{5cm}}}
\subsection*{\en{PVC induced Cardiomyopathy}}
\begin{itemize}
\item افت عملکرد بطن برگشت‌پذیر است.
\item معمولاً فراوانی \en{PVC} باید \en{$>15\%$} باشد در \en{24h}.
\end{itemize}
\subsection*{کاردیومیوپاتی اولیه که \en{PVC} بعداً اضافه شده}
\begin{itemize}
\item \en{Wide QRS}.
\item \en{$\mathrm{PVC}<17\%$}.
\item \en{LV dilation}.
\item \en{multifocal PVC}.
\item مناطق \en{Late Gad Enhancement} در \en{CMR}.
\end{itemize}
اگر شک کنیم که \en{CM} در زمینهٔ \en{PVC} رخ داده است، \en{PVC} حتماً باید درمان شود:
\begin{itemize}
\item بتابلاکر، آمیودارون، \en{Cath Ab}.
\end{itemize}
بیمار با \en{Very frequent PVC} (\en{$>10\text{--}20\%$}) و عملکرد نرمال بطن حتماً باید \en{followup}:
\begin{itemize}
\item سالانه اکو و هولتر ریتم.
\end{itemize}
