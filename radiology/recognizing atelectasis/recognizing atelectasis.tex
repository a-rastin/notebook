\section*{\en{Recognizing Atelectasis}}
\begin{enumerate}[label=\arabic*.,leftmargin=*]
\item آتلکتازی \RAarrow\ از دست رفتن حجم در همه یا بخشی از ریه که منجر به افزایش دانسیتهٔ آن قسمت می‌شود.
  \begin{itemize}
  \item ناشی از یک انسداد: \en{Obstructive atelectasis}
  \end{itemize}
\item یافته‌های تصویربرداری آتلکتازی
  \begin{itemize}
  \item شیفت فیشر اینترلوبار \textbf{به سمت} آتلکتازی
  \item افزایش دانسیتهٔ ریه در ناحیهٔ آتلکتازی
  \item جابجایی ساختارهای متحرک توراکس
    \begin{itemize}
    \item تراشه \RAarrow\ خصوصاً اگر آتلکتازی در \textbf{لوب‌های فوقانی} باشد، به سمت آن شیفت می‌کند.
    \item قلب \RAarrow\ خصوصاً اگر آتلکتازی در \textbf{لوب‌های تحتانی} باشد، به سمت آن شیفت می‌کند.
    \end{itemize}
  \item[$\star$] در حالت طبیعی، بوردور راست قلب باید حدود \en{1 cm} در سمت راست مهره‌ها دیده شود.
  \item همی‌دیافراگم \RAarrow\ سمت درگیر به بالا کشیده می‌شود.
  \item پر شدن بیش از حد ریهٔ سمت مقابل
    \begin{itemize}
    \item ریهٔ پرهوا ممکن است از خط وسط عبور کند.
    \end{itemize}
  \end{itemize}
\end{enumerate}

\subsection*{۳. انواع آتلکتازی}
\subsubsection*{\en{a)} آتلکتازی ساب‌سگمنتال (دیسکوئید یا \en{plate-like})}
\begin{latin}
\begin{itemize}
\item Linear densities extending across all segments of the lower lobes, \textbf{paralleling the diaphragm}
\item In postoperative patients unable to take deep breath
\item Is not caused by bronchial obstruction
\item Most likely related to deactivation of surfactant
\item Disappears within a matter of days
\end{itemize}
\end{latin}

\subsubsection*{\en{b)} آتلکتازی کامپرسیو}
\begin{latin}
\begin{itemize}
\item Loss of volume due to passive compression of the lung
  \begin{itemize}
  \item poor inspiration effort
  \item large pleural effusion
  \item large pneumothorax
  \item Space occupying lesion
  \end{itemize}
\item DDX: Airspace disease at the bases
\end{itemize}
\end{latin}
% The handwritten diagnostic reminder is abbreviated in the original.
شک به آتلکتازی کامپرسیو ریه: بیمار کمتر از ۸ نفس در دقیقه.
\begin{latin}
\noindent In an adult with:
\begin{itemize}
\item Opacified hemithorax
\item No air bronchogram
\item Little or no shift of structures
\end{itemize}
\noindent $\Longrightarrow$ R/O Obstructing bronchogenic carcinoma
\end{latin}

\subsubsection*{\en{c) Round atelectasis}}
\begin{itemize}
\item فرمی از آتلکتازی کامپرسیو
\item معمولاً در پریفری قاعدهٔ ریه
\end{itemize}

\begin{itemize}
\item علت تشکیل
\item ترکیب یک بیماری پلورال قبلی (\en{TB} یا آزبستوز)
\item و یک \en{PE} که آتلکتازی کامپرسیو ایجاد کرده است.
\item زمانی که آتلکتازی ایجاد می‌شود، بیماری پلورال قبلی باعث می‌شود که بخشی از آن گیر بیفتد و به شکل یک توده دیده شود.
  \begin{itemize}
  \item \en{DDX} با توده‌ها و نودول‌ها
  \end{itemize}
\item نمای \en{Comet-tail} در \en{CT}
\end{itemize}

\subsubsection*{\en{d)} آتلکتازی انسدادی (آبزراکتیو)}
\begin{itemize}
\item ناشی از جذب هوا از آلوئول‌ها از طریق مویرگ‌های پولمونری.
\item ۱۸--۲۴ ساعت طول می‌کشد تا یک ریهٔ نرمال، همه کلاپه شود.
  \begin{itemize}
  \item اگر بیمار هوای اتاق تنفس کند.
  \item اگر اکسیژن خالص \RAarrow\ کمتر از ۱ ساعت
  \end{itemize}
\end{itemize}

\subsection*{۴. الگوهای کولاپس در آتلکتازی لوبار}
\begin{latin}
\begin{itemize}
\item Obstructive atelectasis $\longrightarrow$ produces consistently recognizable patterns of collapse.
  \begin{itemize}
  \item Based on $\longrightarrow$ location
  \item degree of force
  \end{itemize}
\item In general
  \begin{itemize}
  \item Lobes collapse in a fan-like configuration
  \item Unaffected lobes will undergo compensatory hyperinflation
    \begin{itemize}
    \item May limit the amount of shift of the mobile chest structures.
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{latin}

\begin{itemize}
\item هرچه یک لوب بیشتر آتلکتاتیک شود، در گرافی ساده کمتر نمایان می‌گردد.
  \begin{itemize}

  \end{itemize}

\begin{latin}
\subsection*{Right upper lobe atelectasis}
\begin{itemize}
\item On the frontal radiograph
  \begin{itemize}
  \item Upward shift of the minor fissure
  \item Rightward shift of the trachea
  \end{itemize}
\item On the lateral radiograph
  \begin{itemize}
  \item Upward shift of the minor fissure
  \item Forward shift of the major fissure
  \end{itemize}
\item[$\star$] The curved edge formed by the mass and the elevated minor fissure comprise what is called the ``\textbf{Reverse S sign of Golden}''.
\end{itemize}

\subsection*{Left upper lobe atelectasis}
\begin{itemize}
\item On the frontal radiograph
  \begin{itemize}
  \item Leftward shift of the trachea
  \item Hazy area of increased density surrounding the left hilum
  \item Elevation with tenting (peaking) of the left hemidiaphragm
  \item Overinflation of the lower lobe
  \end{itemize}
\item On the lateral radiograph
  \begin{itemize}
  \item Forward displacement of the major fissure
  \item Opacified upper lobe forms a band of increased density running roughly parallel to the sternum.
  \end{itemize}
\end{itemize}
\end{latin}

\begin{latin}
\subsection*{Either lower lobe atelectasis}
\begin{itemize}
\item On the frontal radiograph
  \begin{itemize}
  \item Downward shift of the major fissure
  \item Heart may shift toward the side of the volume loss
  \item Elevation of the hemidiaphragm on the affected side
  \item Collapsed lobe forms a triangular density
    \begin{itemize}
    \item Apex at the hilum
    \item Base at the medial portion of the affected hemidiaphragm
    \end{itemize}
  \end{itemize}
\item On the lateral radiograph
  \begin{itemize}
  \item Downward and posterior displacement of the major fissure.
  \end{itemize}
\item[$\star$] In the critically ill patients $\longrightarrow$ Atelectasis occurs most frequently in the \textbf{left lower lobe}.
\end{itemize}
\end{latin}

\noindent $\star$ در آتلکتازی لوب تحتانی چپ \RAarrow\ همه یا بخشی از همی‌دیافراگم چپ سیلوئته می‌شود.

\begin{latin}
\subsection*{Right middle lobe atelectasis}
\begin{itemize}
\item Frontal radiograph
  \begin{itemize}
  \item The minor fissure is displaced downward.
  \item A \textbf{triangular density}
    \begin{itemize}
    \item Base silhouetting the right heart border
    \item Apex pointing toward the lateral chest wall.
    \end{itemize}
  \end{itemize}
\item Lateral radiograph
  \begin{itemize}
  \item The minor fissure displaced inferiorly and the major fissure superiorly
  \item A triangular density
    \begin{itemize}
    \item Apex at the hilum
    \item The base directed anteriorly
    \end{itemize}
  \end{itemize}
\end{itemize}
\end{latin}

\begin{latin}
\subsection*{Entire lung atelectasis}
\begin{itemize}
\item Frontal radiography
  \begin{itemize}
  \item Opacification of the atelectatic lung
  \item Silhouetting of the hemidiaphragm
  \item Shift of the all mobile structures of thorax toward to affected lung.
  \end{itemize}
\item Lateral radiograph
  \begin{itemize}
  \item Silhouetting of the hemidiaphragm
  \item Only one hemidiaphragm on the lateral images.
  \end{itemize}
\end{itemize}
\end{latin}
