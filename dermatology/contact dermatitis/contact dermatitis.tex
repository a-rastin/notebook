\subsection*{درماتیت تماسی تحریکی}
\begin{itemize}
  \item در \cdhl{کودکی} رخ می‌دهد؛ \cdhl{\en{$M<F$}}.
  \item \cdhl{\en{ill defined, Scaly, pink or red patches and plaques}}.
  \item در سطح \cdhl{دورسال دست‌ها} شایع‌تر است.
  \item \cdhl{\en{Diaper dermatitis}}: \en{Spare protected groin folds}.
  \begin{itemize}
    \item ممکن است عفونت ثانویهٔ کاندیدا سوار شود.
    \item \cdhl{نیستاتین} و \cdhl{آزول} موضعی + \cdhl{\en{C/S}} موضعی ضعیف.
  \end{itemize}
\end{itemize}

\subsection*{درماتیت تماسی آلرژیک}
\begin{itemize}
  \item پاسخ ایمنی سلولی، ازدیاد حساسیت تأخیری (نوع \cdhl{\en{IV}}).
  \item \en{Ag} توسط سلول لانگرهانس به \en{T} ارائه می‌شود.
\end{itemize}

\subsubsection*{تظاهر بالینی}
\begin{itemize}
  \item \cdhl{حاد}: \cdhl{درماتیت \en{Rhus}}.
  \begin{itemize}
    \item صورتی روشن، پچ‌های خارش‌دار؛ معمولاً خطی یا با مارجین واضح، داخل پچ وزیکول و بولای شفاف.
    \item \cdhl{\en{7--14} روز} پس از مواجهه؛ مرتبهٔ بعد چند ساعت.
    \item ممکن است هفته‌ها باقی بماند.
  \end{itemize}
  \item \cdhl{مزمن}: صورتی، برجسته، پلاک‌های خارش‌دار.
  \begin{itemize}
    \item \cdhl{درماتیت نیکل}.
    \item \en{DDX} با درماتیت آتوپیک.
  \end{itemize}
  \item \cdhl{تماس مزمن} می‌تواند منجر به درماتیت دائمی شود.
\end{itemize}
