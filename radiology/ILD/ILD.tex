\subsection*{\en{a) Pulmonary interstitial edema}}
\begin{latin}
Because of increased capillary pressure (CHF) or increased capillary permeability (Allergy) or decreased fluid absorption (lymphangitic blockade from metastasis).
\end{latin}
یافته‌های رادیولوژیک:
\begin{latin}
\begin{itemize}
\item fluid in the fissures
\item peribronchial cuffing (\sout{fluid from} fluid in the wall of bronchioles)
\item pleural effusions
\item \textbf{Kerley B lines}
\end{itemize}
\end{latin}
\begin{itemize}
\item پس از درمان ظرف \en{48h} رفع می‌شود.
\end{itemize}
\subsection*{\en{b) Interstitial pneumonia}}
\subsubsection*{\en{b-1) Usual interstitial pneumonia (UIP)}}
\begin{latin}
$\sim$ Idiopathic interstitial fibrosis
$\longleftarrow$ End-stage of this disease
\end{latin}
\begin{itemize}
\item در مردان مسن شایع‌تر است.
\item مرتبط با سیگار و \ILDhl{\en{GERD}}.
\item رادیولوژی قفسهٔ سینه نرمال است، مگر پیشرفته شود.
\end{itemize}
\begin{latin}
\begin{itemize}
\item Fine reticular pattern at the lung bases $\Longrightarrow$ \textbf{Honeycomb lung} $\Longrightarrow$ \textbf{progressive volume loss}.
\item \textbf{Traction bronchiectasis} may be present.
\end{itemize}
\end{latin}
\subsubsection*{\en{b-2) Nonspecific interstitial \sout{fibrosis} pneumonia (NSIP)}}
\begin{latin}
\begin{itemize}
\item In patients with connective tissue diseases
\end{itemize}
\end{latin}
\begin{itemize}
\item \ILDhl{سیستمیک اسکلروزیس}
\end{itemize}
\begin{latin}
\begin{itemize}
\item Better prognosis than the UIP.
\item On CT basilar \textbf{ground-glass opacities} and \textbf{traction bronchiectasis}.
\end{itemize}
\end{latin}

\subsection*{\en{b) Metastases to the lung}}
\subsubsection*{\en{b-1) Hematogenous metastases}}
\begin{latin}
\begin{itemize}
\item 2 or more nodules in the lung
\item When they achieve a large size = Cannonball metastases
\end{itemize}
\end{latin}
\begin{itemize}
\item برست، کولورکتال، \en{RCC}، مثانه، ریه، ملانوم، سارکوم، کارسینوم سرویکوترین.
\end{itemize}
\subsubsection*{\en{b-2) Lymphangitic carcinoma}}
\begin{latin}
\begin{itemize}
\item Resemble pulmonary interstitial edema from CHF
\item It tends to be localized
\item Findings $\longrightarrow$ \textbf{Kerley lines}
\item Fluid in the fissures
\item Pleural effusion
\end{itemize}
\end{latin}
\begin{itemize}
\item برست، ریه، معده، پانکراس، پروستات.
\end{itemize}
