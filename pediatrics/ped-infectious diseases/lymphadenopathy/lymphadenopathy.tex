\begin{itemize}

\item \textbf{سایز طبیعی لنف‌نود:}
  \begin{itemize}
  \item زیر ۱ سانتی‌متر.
  \item ۱۵ میلی‌متر \LymphArrow{} اینگوینال.
  \item ۵ میلی‌متر \LymphArrow{} اپی‌تروکلئار.
  \item ۲ میلی‌متر \LymphArrow{} سوپراکلاویکولار.
  \end{itemize}
\item لنفادنوپاتی یعنی بزرگ شدن لنف‌نودها:
  \begin{itemize}
  \item جنرالیزه \LymphArrow{} حداقل ۲ لنف‌نود غیرمتوالی.
  \item رژیونال.
  \end{itemize}
\item لنفادنیت \LymphArrow{} التهاب حاد یا مزمن لنف‌نود.
\item ویروس‌ها که لنفادنوپاتی جنرالیزه می‌دهند:
  سرخجه، سرخک، آدنوویروس،
  \lr{EBV, CMV, HIV, HBV, HCV}.
\item باکتری‌ها که لنفادنوپاتی جنرالیزه می‌دهند:
  بروسلا، اندوکاردیت، \lr{TB}، سفلیس، لپتوسپیروز.
\item لنفانژیت:
  \begin{itemize}
  \item التهاب کانال‌های لنفی زیرپوستی (\lr{SC}).
  \item معمولاً به دنبال عفونت باکتریایی حاد.
  \item استاف اورئوس، استرپ گروه \lr{A}.
  \end{itemize}
\end{itemize}

\subsection*{لنفادنیت سرویکال}
\begin{itemize}
\item \textbf{شایع‌ترین محل لنفادنیت در کودکان}.
\item فارنژیت \LymphArrow{} \textbf{استرپ گروه \lr{A}}.
\item \lr{EBV}، خراش گربه (بارتونلا)، \lr{TB}، اکتینومایس.
\item نکات مهم در معاینهٔ لنف‌نود:
  اندازه، شکل، تعداد، بافت، سفتی، موبیلیتی، تندرنس، گرمی،
  فلوکچویشن، چسبندگی، مجاورت.
\end{itemize}

\subsection*{مونونوکلئوز عفونی}
\begin{itemize}
\item \lr{EBV}: \textbf{شایع‌ترین علت}.
\item تب، ضعف، خستگی، \textbf{لنفادنوپاتی} گردن یا جنرالیزه،
  تونسیلیت، فارنژیت.
\item \textbf{اگزودا} ته حلق، پتشی ته حلق.
\item لنفادنوپاتی معمولاً \textbf{دوطرفه} است.
\item \textbf{اسپلنومگالی} در ۵۰\% بیماران.
\item هپاتومگالی در ۲۰\% بیماران.
\item راش اریتماتو منتشر در ۸۰\% بیماران
  \LymphArrow{} \textbf{اگر با آموکسی‌سیلین درمان کنیم}.
\item ویروس \lr{EBV}، لنفوسیت‌های \lr{B} را آلوده می‌کند؛
  در غدد بزاقی باقی می‌ماند.
\end{itemize}

\subsection*{سندرم شبه مونونوکلئوز}
\begin{itemize}
\item \lr{CMV}، آدنوویروس، \lr{HBV, HCV, HIV}، توکسوپلاسما گوندی.
\item \lr{CMV} \LymphArrow{} فارنژیت خفیف‌تر، اسپلنومگالی شدیدتر.
\end{itemize}

\subsection*{خراش گربه}
\begin{itemize}
\item بارتونلا (باسیل گرم‌منفی)، بچه‌گربهٔ زیر ۱ سال.
\item پاپول‌های پوستی، گرانولوم ملتحمه \LymphArrow{} در محل خراش.
\item \textbf{لنفادنوپاتی دردناک} (۱–۴ ماه).
\item عوارض نادر: اریتم نودوزوم، استئولیز، انسفالیت، اندوکاردیت، پریکاردیت.
\end{itemize}

\subsection*{برخورد با لنفادنوپاتی}
\begin{itemize}
\item \lr{CBC, CRP, ESR}.
\item \lr{CBC} در مونونوکلئوز عفونی:
  لنفوسیتوز، \textbf{لنفوسیت آتیپیک}، \textbf{ترومبوسایتوپنی}؛
  آنزیم‌های کبدی \lr{\(\uparrow\)}.
\item \textbf{کشت اگزودای لوزه و ضایعات پوستی ضروری است}.
\item اگر شواهد باکتریمی \LymphArrow{} \lr{BC}.
\end{itemize}

\subsection*{اندیکاسیون آسپیراسیون لنف‌نود}
\begin{latin}
Acutely inflamed, fluctuated cervical lymph node,
Specially larger than \textbf{3 cm}, not responding to AB
\end{latin}

\begin{itemize}
\item \textbf{اگر شک به لنفوم} \LymphArrow{} \textbf{بیوپسی}.
  \par \lr{firm, matted, non tender}، یافته‌های سیستمیک.
\item هدف \lr{AB} امپریکال \LymphArrow{} استاف اورئوس، استرپ گروه \lr{A}.
\item لنفادنیت مایکوباکتریایی غیرتوبرکولوزی
  \LymphArrow{} درمان قطعی: بیوپسی اکسیژنال.
\end{itemize}

\subsection*{\lr{DDX} برای لنفادنوپاتی جنرالیزه}
\begin{itemize}
\item \lr{JIA, SLE}، سرم سیکنس، عارضهٔ دارویی (فنی‌توئین و ضدتشنج‌ها).
\item نئوپلاسم‌ها:
  \begin{itemize}
  \item لنفادنوپاتی بدون درد، سفت، \lr{matted}، غیرمتورم.
  \item لوکمی، لنفوم، نوروبلاستوم.
  \end{itemize}
\end{itemize}

\begin{itemize}
\item کودک با لنفادنوپاتی گردنی بدون علائم التهاب حاد
  \LymphArrow{} معمولاً ظرف \textbf{۲–۳ هفته} رفع می‌شود و نیاز به درمان خاصی ندارد.
\end{itemize}

\subsection*{درمان \lr{AB} امپریکال}
\begin{itemize}
\item پنی‌سیلین مقاوم به پنی‌سیلیناز \LymphArrow{} \textbf{اگزاسیلین}.
\item سفالوسپورین نسل \lr{I} \LymphArrow{} سفازولین.
\item \textbf{کلیندامایسین} \LymphArrow{} آلرژیک؛ \lr{S.} اورئوس مقاوم.
\item عدم پاسخ بالینی ظرف \textbf{۴۸–۷۲ ساعت}
  \LymphArrow{} اندیکاسیون بررسی بیشتر.
\end{itemize}
