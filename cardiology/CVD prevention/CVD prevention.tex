\CVDpage{۱}
\subsection*{هفت اصل مهم پیشگیری}
\begin{itemize}
\item سیگار نکشیدن: حداقل یک سال از ترک بگذرد.
\item \en{BMI}: \en{$BMI<25$}.
\item فعالیت فیزیکی: \en{$\geq150$} دقیقه در هفته فعالیت متوسط، \en{$\geq75$} دقیقه شدید.
\item رژیم غذایی.
\item کلسترول: توتال کلسترول \en{$<200\,\mathrm{mg/dl}$}.
\item قند خون: \en{$FBS<100\,\mathrm{mg/dl}$}.
\item فشار خون: \en{$BP\leq120/80$}.
\end{itemize}
\begin{latin}
\noindent CAC $\longrightarrow$ Coronary Artery Calcium
\end{latin}
از روش‌های تصویربرداری مثل آنژیوگرافی عروق کرونری برای بیمارانی که در \en{ASCVD} ریسک ۱۰ ساله بین \underline{۵–۱۰\%} دارند استفاده می‌کنیم اگر به هر دلیل تمایلی برای دادن استاتین نداشته باشیم.

اگر بیمار \underline{\en{lower risk}} باشد ولی \underline{سابقهٔ خانوادگی} بیماری کرونری در سن پایین داشته باشد نیز از روش‌های تصویربرداری (\en{CAC}، آنژیوگرافی) استفاده می‌کنیم.

\begin{itemize}
\item \en{CAC} هر \underline{۵ سال} اعتبار دارد.
\item اگر بیمار استاتین می‌گیرد، تصویربرداری با \en{CAC} \underline{دیگر جایگاهی ندارد}.
\end{itemize}

\CVDpage{۲}
\begin{itemize}
\item در بیمار با \en{CP} پایدار \en{$\longleftarrow$} \en{CCTA} انجام می‌دهیم.
\item \en{CCTA} \en{$\longleftarrow$} در بیماران \en{high Risk} جهت پیشگیری ثانویه.
  \begin{itemize}
  \item برتری آن بر \en{MPI}، \en{SPECT} مشخص نیست.
  \end{itemize}
\end{itemize}
\subsection*{اندیکاسیون درمان با استاتین، \en{COR I}}
\begin{itemize}
\item بیمار با پیشگیری \underline{ثانویه} با سابقهٔ \en{ACS}، \en{MI}، \en{TIA}، بیماری عروق محیطی.
  \begin{itemize}
  \item حداکثر دوز قابل تحمل استاتین یا غیر استاتین برای کاهش \en{LDL} به میزان \underline{حداقل ۵۰\%}.
  \end{itemize}
% Source correction: «ثانویه» is crossed out and replaced by «اولیه».
\item بیمار با پیشگیری اولیه با کلسترول \underline{\en{$LDL\geq190\,\mathrm{mg/dl}$}} (پرایمری هایپرکلسترولمی).
  \begin{itemize}
  \item کاهش \en{LDL} \underline{حداقل ۵۰\%}.
  \end{itemize}
\item بیمار با پیشگیری \underline{اولیه} بدون \en{ASCVD} با \en{RF} (دیابت، سن ۴۰–۷۵، \en{LDL} بین ۷۰–۱۸۹).
  \begin{itemize}
  \item با دوز متوسط / اگر \en{RF} متعدد با دوز بالا.
  \end{itemize}
\end{itemize}
در همهٔ افراد بدون علامت، بین \underline{۴۰–۷۵ سال} هر \underline{۵ سال} ریسک ۱۰ ساله باید اندازه‌گیری شود.

\CVDpage{۳}
\subsection*{ریسک ۱۰ ساله}
\begin{itemize}
\item پایین: \en{$<5\%$}.
\item مرزی: \en{5\%--7.5\%}.
\item متوسط: \en{7.5\%--20\%}.
\end{itemize}
\noindent مرزی و متوسط \en{$\longleftarrow$} \en{$CAC\pm$} \en{$\longleftarrow$} تصمیم برای استاتین.
\begin{itemize}
\item بالا: \en{$>20\%$} \en{$\longleftarrow$} قطعاً استاتین.
\end{itemize}
\subsection*{مواردی که ریسک \en{CVD} را بالا می‌برند ولی در ریسک ۱۰ ساله نیستند}
\begin{itemize}
\item سابقهٔ خانوادگی \en{ASCVD} زودرس (\en{M} زیر ۵۵، \en{F} زیر ۶۵).
\item هایپرکلسترولمی (\en{$LDL\text{-}c\geq160$}، \en{$non\text{-}HDL\text{-}c\geq190$}).
\item \en{CKD} (\en{eGFR 15--49}).
\item سندرم متابولیک.
\item منوپوز زودرس، سابقهٔ پره‌اکلامپسی.
\item وضعیت التهاب مزمن (\en{RA}، لوپوس، پسوریازیس، \en{HIV}).
\item \en{$TG\geq175$} در حداقل ۲ آزمایش.
\item \en{$hsCRP\geq2\,\mathrm{mg/l}$}.
\item \en{$ABI<0.9$}.
\item آپولیپوپروتئین \en{B}: \en{$\geq130\,\mathrm{mg/dl}$}.
\item لیپوپروتئین \en{a}: \en{$\geq50\,\mathrm{mg/dl}$}.
\end{itemize}

\CVDpage{۴}
درمان طولانی‌مدت با استاتین \underline{ایمن} است.
\begin{itemize}
\item استاتین‌ها \en{$\longleftarrow$} مهار سنتز کلسترول با مهار \en{HMG CoA} ردوکتاز.
  \begin{itemize}
  \item اثرات \underline{ضدالتهابی} مستقل از اثر کاهش \en{LDL} دارد.
  \item \en{LDL} را بیشتر از \en{TG} پایین می‌آورد.
  \item عوارض جانبی: درد عضله، اختلال متابولیسم گلوکز، استروک هموراژیک.
  \item ۱۵\% دریافت‌کنندگان میالژی پیدا می‌کنند ولی \en{CK} نرمال است.
  \item میوزیت در ۱\% بیماران که \en{CK} بالا می‌رود.
  \end{itemize}
\end{itemize}
\begin{itemize}
\item قبل از شروع درمان با استاتین \en{$\longleftarrow$} \underline{پنل لیپید}، \underline{\en{CK}}، \underline{\en{AST}}.
\item \underline{۱–۳ ماه} بعد پنل لیپید مجدد اندازه‌گیری شود.
\item فالوآپ دوره‌ای براساس تشخیص پزشک.
\item هدف درمان رسیدن به عدد مشخص نیست بلکه \underline{کاهش حداقل ۵۰\%} میزان \en{LDL} است.
\end{itemize}

\CVDpage{۵}
\begin{itemize}
\item آسپیرین \en{low dose} \en{$\longleftarrow$} پیشگیری ثانویهٔ \en{CVD}.
  \begin{itemize}
  \item نقش آن در پیشگیری اولیه \en{straight forward} نیست.
  \item باید براساس بیمار تصمیم گرفته شود.
  \end{itemize}
\item در بیماران \en{DM} با ریسک فاکتورهای دیگر \en{CVD} توصیه می‌شود.
\item بیماران \underline{۴۰–۷۰ سال} که ریسک افزایش‌یافتهٔ خونریزی ندارند ولی ریسک افزایش‌یافتهٔ \en{CVD} دارند.
\end{itemize}
