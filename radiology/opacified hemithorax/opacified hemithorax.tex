\section*{\en{Opacified Hemithorax}}
\begin{enumerate}[label=\arabic*.,leftmargin=*,itemsep=0.45em]
\item علل همی‌توراکس اوپاک
  \begin{itemize}[nosep]
  \item آتلکتازی تمام یک ریه
  \item یک \en{PE} بزرگ
  \item پنومونی تمام یک ریه
  \item پنومونکتومی (نادر)
  \end{itemize}

\item آتلکتازی تمام یک ریه
  \begin{itemize}[nosep]
  \item به علت انسداد کامل برونش اصلی چپ یا راست.
  \item علت شایع براساس سن:
    \begin{itemize}[nosep]
    % The far left of this handwritten branch touches the photograph edge.
    \item افراد مسن \OHarrow\ بدخیمی \OHarrow\ کارسینوم برونکوژنیک
    \item افراد جوان \OHarrow\ موکوس پلاگ بر اثر آسم
    \item کودکان \OHarrow\ آسپیراسیون جسم خارجی
    \item بیماران شدیداً بیمار \OHarrow\ موکوس پلاگ
    \end{itemize}
  \end{itemize}

\item \en{PE} شدید
  \begin{itemize}[nosep]
  \item می‌تواند به علت تجمع اگزودا، ترانسودا یا خون باشد.
  \item شدت زیاد مایع منجر به فشرده شدن ریه می‌شود.
  \item بدخیمی علت شایع است:
    \begin{itemize}[nosep]
    \item کارسینوم برونکوژنیک
    \item متاستاز
    \end{itemize}
  \item \en{CHF} معمولاً \en{PE} آن‌قدر شدیدی نمی‌دهد که منجر به آتلکتازی شود.
  \end{itemize}
\end{enumerate}

  \begin{itemize}[itemsep=0.35em]

  \item ظرف دو هفته فضای خالی با مایع پر می‌شود.
  \item ظرف ۴ ماه فیبروز تشکیل می‌شود.
  \item قلب و تراشه به طرف آن شیفت پیدا می‌کنند.
  \item معاینهٔ فیزیکی مشابه آتلکتازی است.
  \end{itemize}
