\section*{۱۵۲ ـ \BCTen{Blood component therapy}}
\subsection*{واکنش‌های ترانسفیوژن}

\subsubsection*{\BCTen{Acute Hemolytic Transfusion Reaction}}
\begin{itemize}
  \item علائم: \BCThl{شوک}، \BCThl{درد کمر}، تب، همولیز داخل عروق، هموگلوبینمی، هموگلوبین‌اوری (۵ روز بعد).
  \item درمان: قطع ترانسفیوژن، مایع‌درمانی، اصلاح هایپرکالمی.
\end{itemize}

\subsubsection*{\BCTen{Delayed Hemolytic Transfusion Reaction}}
\begin{itemize}
  \item علائم: شروع \BCThl{\BCTen{7--14} روز بعد}؛ درد، تب، زردی، هموگلوبین‌اوری، افت هموگلوبین، رتیکولوسایتوز.
  \item درمان: فقط \BCTen{RBC} آنتی‌ژن‌منفی دریافت کند.
\end{itemize}

\subsubsection*{\BCTen{Febrile nonhemolytic Transfusion Reaction}}
\BCThl{۱ به هر ۱۰۰ مورد}
\begin{itemize}
  \item علائم: \BCThl{تب ظرف ۴ ساعت}، لرز.
  \item درمان: تب‌بر، کشت خون، آنتی‌بیوتیک، عفونت.
\end{itemize}

\subsubsection*{\BCTen{Allergic Transfusion Reaction}}
\BCThl{۱ به هر ۱۰۰ مورد}
\begin{itemize}
  \item علائم: \BCThl{کهیر}، خارش، راش ماکولوپاپولر، ادم، دیسترس تنفسی، افت \BCTen{BP}؛ \BCThl{ظرف ۴ ساعت}.
  \item درمان: دیفن‌هیدرامین + هیدروکورتیزون.
\end{itemize}

\subsection*{آنمی تغذیه‌ای و \BCTen{Packed cell}}
\begin{itemize}
  \item \BCThl{برای درمان آنمی تغذیه‌ای بدون علامت} از \BCThl{\BCTen{packed cell} استفاده نمی‌کنیم}؛ خوراکی درمان می‌کنیم.
  \item \BCTen{Packed cell}: مقدار \BCTen{10--15 mL/kg}؛ افزایش \BCTen{2--3 g/dL} عدد هموگلوبین.
\end{itemize}
