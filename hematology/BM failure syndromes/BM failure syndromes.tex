\section*{\BMEn{Bone Marrow Failure Syndromes}}
\BMEn{Harrison -- 102}\hfill صفحهٔ ۱

\begin{enumerate}
\item بیماری‌های نارسایی مغز استخوان:
  \begin{itemize}
  \item \BMEn{Aplastic Anemia}
  \item \BMEn{Myelodysplastic syndrome (MDS)}
  \item \BMEn{Pure red cell aplasia (PRCA)}
  \item \BMEn{Myelophthisis}
  \end{itemize}
\item آنمی هایپوپرولیفراتیو ویژگی اصلی این اختلالات است.
  \begin{itemize}
  \item ولی پان‌سایتوپنی اختلال شایع‌تری است.
  \end{itemize}
\item در نارسایی مغز استخوان علت کم بودن سلول‌های خونی:
  \begin{itemize}
  \item \BMEn{Deficient Hematopoiesis}
  \end{itemize}
\item علل اختلال مغز استخوان:
  \begin{itemize}
  \item عفونت
  \item التهاب
  \item بدخیمی
  \end{itemize}
\item سندرم‌های نارسایی هماتوپوئتیک بر اساس مورفولوژی غالب مغز استخوان دسته‌بندی می‌شوند.
\end{enumerate}
