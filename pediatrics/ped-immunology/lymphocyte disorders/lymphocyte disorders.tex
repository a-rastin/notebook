\subsection*{بیماری‌های کمبود آنتی‌بادی}
باکتری‌های کپسول‌دار، مایکوپلاسما، برخی ویروس‌ها.

\subsection*{۱. آگاماگلوبولینمی}
\begin{itemize}
\item \textbf{نبود یا ضعف عملکرد \lr{B cells}} \LDarrow{} سطح پایین \lr{Ab}.
\item آگاماگلوبولینمی وابسته به \lr{X (XLA)} \LDarrow{} ۸۰\% موارد مادرزادی.
\item کمبود \lr{B cell}، پان هایپوگاما، \textbf{عدم وجود بافت لنفاوی}.
\item آگاماگلوبولینمی \lr{AR} \LDarrow{} شبیه \lr{XLA} در بالین.
\item \textbf{شروع علائم ۴ تا ۱۲ ماه بعد از تولد}.
\item عفونت‌های شایع: \textbf{اوتیت، پنومونی}، هموفیلوس تیپ \lr{B}،
انتروویروس، مایکوپلاسما، سودومونا.
\end{itemize}
\begin{latin}
Vaccine associated poliomyelitis (oral Vaccine)
\end{latin}

\subsection*{۲. \lr{CVID}}
\begin{latin}
Common Variable immunodeficiency
\end{latin}
\begin{itemize}
\item \textbf{هایپوگاماگلوبولینمی پس از یک دورهٔ سیستم ایمنی \lr{N/L}}.
\item شایع‌ترین زمان: \textbf{دههٔ ۲ و ۳ زندگی}.
\item \textbf{\lr{Serum IgG < 500}}، معمولاً زیر ۳۰۰؛ \lr{\(IgA<10\)}.
\item \lr{IgM} می‌تواند پایین باشد یا نباشد.
\item \lr{Ab} علیه پلی‌ساکارید مثل پنوموکوک \LDarrow{} \textbf{تقریباً هیچ‌وقت وجود ندارد}.
\item تعداد \lr{B-cell} متفاوت است ولی همیشه وجود دارند
\LDarrow{} افتراق از آگاماگلوبولینمی مادرزادی.
\item \textbf{اسپلنومگالی و تانسیل‌های بزرگ} \LDarrow{} ممکن است وجود داشته باشد.
\item مستعد عفونت‌های تنفسی و \lr{GI}.
\item عوارض خودایمنی، مهم‌ترین علت مرگ \textbf{غیرعفونی} در این بیماران.
\item سایتوپنی خودایمن، بدخیمی‌ها (لنفوم \lr{B} سل)، \lr{ILD}،
مولتی‌سیستم گرانولوماتوزیس، هایپرپلازی کبد، \lr{IBD}.
\end{itemize}

\LDpage{۲}
\subsection*{۳. \lr{Selective IgA deficiency}}
\begin{itemize}
\item \textbf{\lr{\(IgA<10\)} سرم} ولی بقیهٔ \lr{Ig}ها \lr{N/L} هستند.
\item برای تأیید تشخیص، سن بیمار باید \textbf{حداقل ۴ سال} باشد.
\item \textbf{اغلب بیماران بدون علامت هستند}.
\item عفونت‌های سینوپولمونری راجعه.
\item درمان: \textbf{\lr{AB} پروفیلاکسی}؛ در موارد نادر دادن \lr{Ab}.
\item افزایش ریسک آتوپی، خودایمنی، سلیاک.
\item در برخی موارد وراثت اتوزومال.
\end{itemize}

\subsection*{۴. \lr{IgG subclass deficiency}}
\begin{itemize}
\item \textbf{\lr{IgG} توتال \lr{N/L} است} ولی یک یا چند ساب‌کلاس از ۴ ساب‌کلاس پایین هستند.
\item \textbf{عفونت شدید} یا راجعه می‌تواند رخ دهد.
\end{itemize}

\subsection*{۵. \lr{Transient hypogama. of infancy}}
\begin{itemize}
\item وقتی که هایپوگاما فیزیولوژیک شیرخوار طول بکشد.
\item \lr{\(Ig<200\)} سرم در ۶ ماهگی که در حالت طبیعی تا
\LDrange{2}{4} سالگی باید \lr{N/L} شود.
\item افزایش عفونت‌های سینوپولمونری.
\item \textbf{سطح سلول‌های \lr{B} و \lr{T} طبیعی است}؛ پاسخ \lr{Ab}ها نیز طبیعی است.
\item ممکن است گذرا نباشد و ماندگار شود.
\end{itemize}

\subsection*{۶. \lr{Specific Ab deficiency syndrome}}
\begin{itemize}
\item \textbf{بعد از ۲ سالگی}.
\item عفونت راجعه در عین \textbf{سطح \lr{Ig} طبیعی} و \textbf{تعداد لنفوسیت طبیعی}.
\item \textbf{ناتوانی در تولید \lr{Ab} علیه یک \lr{Ag} خاص}.
\item ریسک عفونت‌های سینوپولمونری.
\item \lr{AB} پروفیلاکسی یا \lr{Ab} می‌دهیم.
\end{itemize}

\LDpage{۳}
\subsection*{\lr{Combined immunodeficiency diseases}}
تولید یا عملکرد سلول \lr{T} مختل است.

\subsection*{۱. \lr{X-linked hyper IgM syndrome}}
\begin{itemize}
\item شایع‌ترین فرم سندرم \lr{hyper IgM}.
\item \textbf{نقص در لیگاند \lr{CD40}} \LDarrow{} اختلال عملکرد سلول \lr{T}.
\end{itemize}

\subsection*{۲. \lr{AR hyper IgM syndrome}}
\begin{itemize}
\item اگر نقص در خود \lr{CD40} باشد.
\item شبیه \lr{X-linked} در بالین.
\end{itemize}

\subsubsection*{سندرم هایپر \lr{IgM}}
\begin{itemize}
\item ایزوتایپ‌های \lr{IgM} و \lr{IgD} نمی‌توانند به \lr{IgG}، \lr{A}، \lr{E}
سوئیچ کنند \LDarrow{} \textbf{\lr{lack of memory}}.
\item \textbf{سطح \lr{IgG} پایین است یا کلاً وجود ندارد}؛ \lr{A} و \lr{E} نیز.
\item افزایش ریسک عفونت‌های سینوپولمونری.
\end{itemize}

\subsection*{۳. \lr{Severe Combined Immunodeficiency (SCID)}}
\begin{itemize}
\item \textbf{تعداد یا عملکرد} سلول \lr{T} به‌شدت مختل است.
\item بالین: \lr{FTT}، عفونت باکتریال شدید، کاندیدیازیس مزمن، عفونت قارچی،
عفونت‌های ویروسی مزمن.
\item فرآورده‌های خونی باید \textbf{اشعه‌دیده}، کم‌لکوسیت، \lr{CMV} فری باشند.
\item \lr{GVHD} خیلی خطرناک است برای این‌ها.
\item شایع‌ترین فرم آن \textbf{\lr{X-linked}}.
\item هیچ سلول \lr{T} و \lr{NK} وجود ندارد ولی تعداد \lr{B} طبیعی است.
\item \textbf{سطح \lr{Ig} بسیار پایین است}.
\item \lr{SCID} اتوزومال مغلوب (\lr{AR SCID}):
\textbf{هیچ سلول \lr{T}} ولی \textbf{\lr{NK} و \lr{B} طبیعی هستند}.
\end{itemize}

\LDpage{۴}
\subsection*{۴. سندرم \lr{Omenn}}
\begin{itemize}
\item \textbf{واریانتی از \lr{SCID}}.
\item لنفادنوپاتی، اگزفولیاتیو اریترودرما، \textbf{هپاتواسپلنومگالی}،
\textbf{ائوزینوفیلی شدید}، افزایش \lr{IgE} سرم، اختلال عملکرد \lr{T} سل.
\item در خون محیطی \textbf{\lr{T} سل وجود دارد ولی عملکرد ندارد}.
\end{itemize}

\subsection*{۵. کمبود آدنوزین دآمیناز (\lr{ADA}) و پورین نوکلئوزید فسفوریلاز}
\begin{itemize}
\item منجر به \lr{SCID}.
\item \textbf{تجمع موادی که برای لنفوسیت‌ها سمی است}؛ هم \lr{T} و \lr{B} و \lr{NK}.
\item \textbf{عفونت‌های شدید در اوایل زندگی}.
\item اگر عملکرد آنزیم‌ها نسبی باشد ممکن است تا ۵ سالگی یا حتی بلوغ تشخیص داده نشود.
\end{itemize}

\subsection*{۶. \lr{Bare lymphocyte syndrome}}
\begin{itemize}
\item اختلال در تنظیم بیان \lr{MHC II} یا \lr{MHC I}.
\item \textbf{بافت لنفاوی و سلول‌های \lr{B} طبیعی}.
\item سلول \lr{T}، \lr{CD4} به‌شدت کم \LDarrow{} اگر \lr{MHC II} کم باشد.
\item \lr{CD8} به‌شدت کم \LDarrow{} اگر \lr{MHC I} کم باشد.
\item ممکن است تعداد سلول‌های \lr{T} طبیعی ولی عملکرد نداشته باشند.
\end{itemize}

\subsection*{۷. سندرم دی‌جرج}
\begin{itemize}
\item (سندرم ولوکاردیوفاسیال) \LDarrow{} حذف بخشی از کروموزوم \lr{22q}.
\end{itemize}
\begin{latin}
CATCH 22 syndrome:
\begin{itemize}
\item Cardiac anomalies
\item Abnormal facies
\item Thymic hypoplasia
\item Cleft plate
\item Hypocalcemia
\end{itemize}
\end{latin}
\begin{itemize}
\item تتانی ناشی از هیپوکلسمی.
\item \textbf{آنومالی‌های قوس آئورت}.
\item افزایش ریسک عفونت.
\item تعداد و عملکرد سلول‌های \lr{T} پایین است.
\item \textbf{با افزایش سن بهتر می‌شود}.
\item در موارد شدید شبیه \lr{SCID}.
\end{itemize}

\LDpage{۵}
\subsection*{۸. سندرم \lr{Wiskott-Aldrich}}
\begin{itemize}
\item وابسته به \lr{X}.
\item درمان: پیوند هماتوپوئتیک استم‌سل.
\item \textbf{\lr{1/3} مرگ‌ها به علت خونریزی؛ \lr{2/3} عفونت}.
\item \textbf{ترومبوسایتوپنی}، اگزما، نقص ایمنی سلولی و هومورال، استعداد ابتلا
به بیماری‌های لنفوپرولیفراتیو.
\item \textbf{سطح \lr{IgE} و \lr{IgA} بالا}؛ \textbf{سطح \lr{IgM} پایین}.
\item \textbf{سایتوپنی خودایمن} وجود دارد؛ پاسخ به کپسول‌دارها ضعیف است.
\end{itemize}

\subsection*{۹. ترومبوسایتوپنی وابسته به \lr{X} ایزوله}
نقص ژنی مشابه ویسکوت ـ آلدریچ است.

\subsection*{۱۰. آتاکسیا ـ تلانژکتازیا}
\begin{itemize}
\item جهش ژن \lr{ATM} (شناسایی آسیب \lr{DNA} و چک‌پوینت سلولی).
\item تلانژکتازی پوستی و ملتحمهٔ چشم.
\item \textbf{آتاکسی مخچه‌ای پیشرونده}، تخریب سلول‌های پورکنژ.
\item \lr{IgA deficiency}، \lr{IgG\(_2\) subclass def.}، \lr{low IgE level}،
\textbf{افت عملکرد سلول \lr{T}}.
\item ممکن است منجر به \textbf{لوکمی و لنفوم} شود.
\item حساس به \lr{RT} است.
\item ریسک \lr{ILD} بالا، تأخیر در بلوغ جنسی وجود دارد.
\item درمان: \lr{AB} و \lr{IVIG}.
\end{itemize}

\subsection*{۱۱. \lr{APECED}}
(کاندیدیازیس جلدی مخاطی مزمن)؛ (اتوایمون پلی‌اندوکرینوپاتی).
\begin{itemize}
\item \textbf{۳ تا ۵ سالگی}.
\item درگیری غشاهای مخاطی، پوست، ناخن.
\item تولید \lr{Ab} طبیعی است ولی \textbf{پرولیفراسیون لنفوسیت رخ نمی‌دهد}.
\item به درمان ضدقارچ موضعی جواب نمی‌دهد.
\item در اکثر بیماران یک بیماری اندوکرین خودایمنی در کودکی رخ می‌دهد.
\item \textbf{هایپوپاراتیروئیدی، آدیسون}.
\item پس از تشخیص \LDarrow{} ارزیابی دوره‌ای از نظر بیماری‌های اندوکراین خودایمنی.
\item به دلیل جهش ژن، لنفوسیت‌های حساس به بافت‌های اندوکراین در تیموس حذف نمی‌شوند.
\end{itemize}

\LDpage{۶}
\subsection*{۱۲. بیماری لنفوپرولیفراتیو وابسته به \lr{X}}
\begin{itemize}
\item \textbf{پاسخ ایمنی غیرتنظیم‌شده به \lr{EBV}}.
\item بیمار هیچ مشکلی ندارد تا زمانی که به \lr{EBV} مبتلا شود
\LDarrow{} ۸۰\% مرگ.
\item \textbf{نکروز کبد} به دنبال حملهٔ \lr{CD8}.
\item بیمارانی که نجات پیدا می‌کنند \LDarrow{} هایپوگاما شدید و ریسک بالای
\lr{AA} و لنفوم.
\end{itemize}

\subsection*{۱۳. سندرم \lr{Hyper-IgE}}
\begin{itemize}
\item \textbf{\lr{IgE} بسیار بالا در سرم}؛ اکثراً \lr{AD}، بعضی \lr{AR}.
\item \textbf{راش پوستی} شبیه درماتیت آتوپیک، ائوزینوفیلی.
\item آبسه‌های پوستی استافیلوکوکی در ریه، مفصل هم دیده می‌شود.
\item استئوپنی، شکستگی، شلی مفصل، صورت خشن.
\item \textbf{پنوماتوسل‌های غول‌پیکر} در ریه پس از عفونت استرپتوکوکی.
\item با اینکه بقیهٔ \lr{Ig}ها \lr{N/L} هستند ولی \textbf{پاسخ ایمنی هومورال ضعیف است}.
\item درمان \lr{AB} طولانی‌مدت علیه استافیلوکوک.
\end{itemize}

\subsection*{یادداشت‌های پایانی}
\begin{itemize}
\item در بیمار نقص ایمنی هر فرآوردهٔ خونی اشعه‌دیده و بدون \lr{CMV}.
\item \lr{GVHD} مرگ‌آور است.
\item واکسن زنده نباید دریافت کند.
\item غربالگری نوزادان برای \lr{SCID}.
\end{itemize}
