\NCApage{۱}
{\footnotesize\en{Date: Radiology}\quad\en{Subject: \_\_\_}\quad\en{Page}\par}
\begin{enumerate}[label=\arabic*.,leftmargin=*,itemsep=0.5em]
\item تخمین اندازهٔ قلب در \en{frontal CXR} \NCAarrow\ نسبت کاردیوتوراسیک:
  \begin{itemize}
  \item اندازهٔ سایهٔ قلب در عریض‌ترین نقطه / اندازهٔ داخلی قفسهٔ سینه در عریض‌ترین نقطه.
  \item معمولاً در سطح دیافراگم.
  \end{itemize}
\item نسبت کاردیوتوراسیک:
  \begin{itemize}
  \item باید در طی دم کامل \en{CXR} گرفته شود.
  \item \en{Nine Posterior Ribs Showing}.
  \item \en{$50\% > \mathrm{N/L}$}.
  \end{itemize}
\item کانتورهای قلب:
  \begin{itemize}
  \item سمت راست:
    \begin{itemize}
    \item آئورت صعودی.
    \item یک فرورفتگی جزئی محل \en{Double density left atrium}.
    \item دهلیز راست.
    \end{itemize}
  \item سمت چپ:
    \begin{itemize}
    \item \en{Aortic Knob} (قوس آئورت به روی پروگزیمال آئورت نزولی).
    \item \en{Main Pulmonary Artery}.
    \item یک فرورفتگی جزئی محل \en{Left atrial enlargement}.
    \item بطن چپ.
    \end{itemize}
  \end{itemize}
\item \NCAdeleted{قوس آئورت} آئورت صعودی نباید بیش از بوردر راست قلب (دهلیز راست)، سمت راست باشد.
\item \en{Aortic Knob} باید کمتر از \en{35 mm} باشد و معمولاً تراشه را کمی به راست منحرف می‌کند.
\item کانتور شریان ریوی مشترک معمولاً فرورفته یا صاف است ولی در افراد جوان و خانم‌ها به طور نرمال برآمده است.
\end{enumerate}

\NCApage{۲}
{\footnotesize\en{Date: \_\_\_}\quad\en{Subject:} ۴\quad\en{Page}\par}
\begin{enumerate}[label=\arabic*.,leftmargin=*,itemsep=0.5em]
\setcounter{enumi}{6}
\item واکنش بطن‌ها به انسداد خروجی:
  \begin{itemize}
  \item در ابتدا هایپرتروفی.
  \item سپس دیلاتاسیون.
  \item آئورتیک استنوزیس، کوآرکتاسیون آئورت، پولمونیک استنوزیس، \en{HTN} سیستمیک.
  \end{itemize}
\item بزرگ شدن دهلیزها منجر به کاردیومگالی نمی‌شود و کلیهٔ بطن‌ها باید بزرگ شوند.
  \begin{itemize}
  \item در ابتدای \en{MS}، دهلیز چپ بزرگ می‌شود ولی کاردیومگالی نداریم.
  \end{itemize}
\item به طور کلی رایج‌ترین \en{chamber enlargement}ها ناشی از \en{overload} هستند تا ابنورمالیتی فشار داخل حفره.
  \begin{itemize}
  \item به همین دلیل بزرگ شدن حفرات بیشتر ناشی از رگورجیتیشن دریچه‌ها هستند تا ناشی از استنوز آن‌ها.
  \end{itemize}
\item \en{CT} اسکن قلب معمولاً با کنتراست و به همراه یک \en{EKG} گرفته می‌شود تا آرتیفکت‌های ناشی از حرکات قلب را کاهش دهد.
\item استنوز آئورت (\en{AS}):
  \begin{itemize}
  \item \en{Post-stenotic dilatation}.
  \item کانتور آئورت صعودی بزرگ می‌شود.
  \item \en{Aortic Knob} و آئورت نزولی نرمال هستند.
  \end{itemize}
\item \en{HTN} سیستمیک:
  \begin{itemize}
  \item کانتور آئورت صعودی و نزولی هر دو بزرگ می‌شوند.
  \item \en{Aortic Knob} بزرگ می‌شود.
  \end{itemize}
\item برای بررسی قلب در \en{CT} آن را در ۶ سطح بررسی می‌کنیم.
\end{enumerate}

\NCApage{۳}
{\footnotesize\en{Note book}\quad\en{Subject:} ۴\quad\en{Veor: \_\_\_\quad Month: \_\_\_\quad Date: \_\_\_}\par}
{\footnotesize حاشیهٔ بالا: \NCAuncertain{\en{AS}، خط‌خورده}\par}
\begin{enumerate}[label=\arabic*.,leftmargin=*]
\setcounter{enumi}{13}
\item سطوح بررسی قلب در \en{CT}:
\end{enumerate}
\subsection*{\en{a) Five-vessel level}}
\begin{latin}
$\longrightarrow$ Trachea 2 cm in diameter
\end{latin}
\begin{itemize}
\item مری در خلف تراشه است و می‌تواند در سمت راست یا چپ آن باشد \NCAarrow\ معمولاً خوابیده.
\item ساختارهای وریدی معمولاً قدامی‌تر از شریان‌ها هستند.
\item ورید براکیوسفالیک چپ (بی‌نام) درست در پشت استرنوم است.
\end{itemize}
\subsection*{\en{b) Aortic arch level}}
\begin{latin}
$\longrightarrow$ Aortic arch + SVC + Azygos vein
\end{latin}
\begin{itemize}
\item ورید آزیگوس به \en{SVC} می‌ریزد.
\item آئورت نزولی حدود \en{5 mm} کوچک‌تر از نزولی.
% «نزولی» occurs twice in the source; retained without medical correction.
\end{itemize}
\subsection*{\en{c) Aortopulmonary Window level}}
\begin{latin}
$\longrightarrow$ Ascending and descending aorta + SVC + uppermost aspect of the Left pulmonary artery (maybe)
\end{latin}
\begin{itemize}
\item درست زیر \en{AA} فضای \en{APW} وجود دارد.
\item بالای شریان ریوی \NCAarrow\ محل شایع درگیری لنف نود.
\item در این محل یا کمی پایین‌تر تراشه دوشاخه می‌شود.
\end{itemize}
\subsection*{\en{d) Main Pulmonary Artery level}}
\begin{latin}
$\longrightarrow$ Main, right, left pulmonary artery + right and left main bronchi + bronchus intermedius
\end{latin}
\begin{itemize}
\item شریان ریوی چپ بالاتر از راست است و ادامهٔ مسیر شریان ریوی اصلی است.
\item شریان ریوی راست با زاویهٔ ۹۰ جدا می‌شود.
\end{itemize}

\subsection*{\en{e) High Cardiac level}}
\begin{latin}
$\longrightarrow$ Left atrium + right atrium + aortic root + right ventricular outflow tract
\par\medskip
$\longrightarrow$ The Right Ventricular Outflow tract lies ALS to the root of aorta.
\par\smallskip
ALS $\longrightarrow$ Ant. / Lat. / Sup.
\end{latin}
\begin{itemize}
\item دریچهٔ پولمونری \en{ALS} نسبت به دریچهٔ آئورت.
\end{itemize}
\subsection*{\en{f) Low Cardiac level}}
\begin{latin}
$\longrightarrow$ RA + RV + LAV + IVS
\end{latin}
% LAV is the source abbreviation, retained as written.
\begin{itemize}
\item بطن راست درست در خلف استرنوم.
\item نسبت به بطن چپ دیواره نازک‌تر و ترابکول‌ها بیشتر.
\item حداکثر ضخامت پریکارد \en{2 mm} است.
\end{itemize}
\begin{enumerate}[label=\arabic*.,leftmargin=*,itemsep=0.5em]
\setcounter{enumi}{14}
\item کلسیم اسکورینگ:
  \begin{itemize}
  \item در \en{CT} بدون کنتراست.
  \item بر اساس میزان و دانسیتهٔ کلسیم موجود در شریان‌های کرونری.
  \item نبود کلسیفیکاسیون ارزش اخباری منفی بالایی برای نبود بیماری انسدادی لومن شریان کرونری دارد.
  \item در بیماران بدون علامت انجام می‌شود.
  \end{itemize}
\item \en{Coronary CT Angiography (CCTA)}:
  \begin{itemize}
  \item برای بیماران با \en{CP} حاد یا مزمن انجام می‌شود.
  \item ارزش اخباری منفی بالایی دارد.
  \item دوز اشعه نسبتاً بالاست.
  \end{itemize}
\end{enumerate}
