\section*{تصویر ۱ ـ صفحهٔ ۱}
\subsection*{\lr{IUGR}}
\begin{itemize}
  \item ناشی از ابنورمالیتی‌های: مادری، جفتی، جنینی.
  \item انحراف از الگوی رشد قابل انتظار.
  \item در اصل نوعی انطباق با شرایط نامناسب داخل رحم است.
  \item شایع‌ترین علل آن، \FDmark{علل مادری} هستند که با زایمان زودرس همراهی دارند.
\end{itemize}

\subsection*{\lr{SGA}}
\begin{itemize}
  \item لزوماً پاتولوژیک نیست.
  \item وزن هنگام تولد نوزاد، زیر صدک دهم و یا دو انحراف معیار کمتر از متوسط باشد.
\end{itemize}

\subsection*{تظاهرات نوزاد \lr{SGA}}
\begin{itemize}
  \item \FDmark{\lr{Wasted appearance}}: ساق و پای نازک.
  \item یافته‌ها: \FDunclear{غیر قرینگی}؛ سر بزرگ‌تر نسبت به بدن: \FDmark{\lr{CNS Sparing}}.
  \item \FDmark{فونتانل قدامی بدن} \FDunclear{خوانش عبارت مشخص‌شده با ماژیک، به‌ویژه واژهٔ آخر}: کاهش چربی زیرپوستی.
  \item بند ناف نازک و \FDmark{آغشته به مکونیوم}.
  \item پوست خشک.
\end{itemize}

ارگانومگالی، زردی، رش پوستی، اختلالات چشمی
(کوریورتینیت، کاتاراکت، گلوکوم، قرنیهٔ \lr{cloudy}):
\begin{itemize}
  \item شک به عفونت‌های مادرزادی.
  \item شک به اختلالات متابولیک نوزاد.
\end{itemize}

جنین‌های \lr{IUGR} و \lr{SGA} به‌ویژه حین زایمان دچار چالش می‌شوند:
\begin{itemize}
  \item اسیدوز متابولیک، آسفیکسی، هایپوکسی، هایپوتنشن، هایپوگلیسمی.
  \item پلی‌سایتمی، \lr{MAS}، \lr{PPH} نوزادان.
\end{itemize}

\FDmark{منیج نوزاد \lr{IUGR} و \lr{SGA} علامتی و حمایتی است.}

\subsection*{داروهایی که منجر به \lr{IUGR} می‌شوند}
\begin{itemize}
  \item \FDmark{متوترکسات} (آنتی‌متابولیت‌ها).
  \item فلزات سنگین (سرب و جیوه).
  \item \FDmark{وارفارین}.
  \item \FDmark{استروئید} (پردنیزولون).
  \item \FDmark{نارکوتیک‌ها}.
  \item هیدانتوئین، الکل، کوکائین.
\end{itemize}

\section*{تصویر ۲ ـ صفحهٔ ۲}
\subsection*{هیدروپس فتالیس}
\begin{enumerate}
  \item ایمن.
  \item غیرایمن.
\end{enumerate}
\begin{itemize}
  \item تجمع شدید مایع در پوست و یک یا چند فضای دیگر.
  \item پلور، پریتوئن، پریکارد، جفت.
  \item \FDmark{عموماً با پلی‌هیدرآمنیوس همراهی دارد.}
  \item \FDmark{ایزوایمیونیزیشن علیه گروه خونی رزوس (\lr{D})}: یکی از علل.
\end{itemize}

\subsection*{علل غیرایمن}
\begin{itemize}
  \item \lr{CHF}.
  \item انسداد جریان لنفاوی.
  \item کاهش فشار آنکوتیک جفت.
  \item \FDmark{عفونت}؛ \lr{\(\downarrow\) pro}.
\end{itemize}

\subsection*{بررسی هیدروپس فتالیس}
به دنبال \underline{علت زمینه‌ای} می‌گردیم.
\begin{itemize}
  \item یافته‌های شایع در مادر: \lr{HTN}، آنمی، بارداری‌های متعدد، جفت حجیم‌شده، ادم، پلی‌هیدرآمنیوس.
  \item یافته‌های شایع جنین: تاکی‌کاردی، آسیت، ادم اسکالپ، پلورال و پریکاردیال افیوژن.
  \item نمونه‌برداری از مایع آمنیوتیک: کاریوتایپ، کشت، \FDmark{\lr{\(\alpha\)-FP}}، آنالیز.
        حاشیهٔ مربوط به آنالیز: آنزیمی، متابولیک.
\end{itemize}

جنین هیدروپس فتالیس معمولاً پس از تولد \underline{نیاز به احیا دارد}:
\begin{itemize}
  \item آسپیراسیون مایع آسیت و پلورال افیوژن.
  \item در صورت آنمی جدی: \FDmark{ترانسفوزیون \lr{P.C}}.
\end{itemize}
