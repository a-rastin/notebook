\section*{\en{Breast Imaging}}
\BInote{۱}{یک زن بدون ریسک فاکتور شناخته‌شده، \en{Lifetime risk} برای کنسر برست: ۱۲\%.}
\BInote{۲}{ریسک فاکتورهای کنسر برست}
\begin{BIpoints}
\item منارک زودرس
\item منوپوز دیررس
\end{BIpoints}
\BInote{۳}{\en{Lifetime risk for breast cancer}}
\begin{BIenglish}
\item Average ($<15\%$): No known risk factor
\item Intermediate (15--25\%)
\begin{itemize}
\item Strong family history of breast cancer
\item No BRCA mutation
\item CKEK2 mutation % spelling as written in the scan
\item Past history of lobular carcinoma in situ or atypical hyperplasia
\item Highest category of mammographic density
\end{itemize}
\item High ($>20\%$)
\begin{itemize}
\item BRCA mutation carriers and their untested first-degree relatives
\item History of chest radiation between age 10--30
\item Personal history of invasive breast cancer or ductal carcinoma in situ (DCIS)
\end{itemize}
\end{BIenglish}
\BInote{۴}{زنان با موتاسیون ژن \en{BRCA}، \en{Lifetime risk}: ۸۰\%؛ تا سن ۹۰ سالگی.}
\BInote{۵}{ریسک سرطان تخمدان}
\begin{BIpoints}
\item \en{BRCA1}: ۵۵\%
\item \en{BRCA2}: ۲۵\%
\end{BIpoints}

\BInote{۸}{سونوگرافی (\en{US})}
\begin{BIpoints}
\item بیشترین کاربرد در ترکیب با ماموگرافی برای تشخیص
\item می‌تواند به عنوان یک روش غربالگری نیز استفاده شود.
\item \textbf{روش بررسی اولیه زنان با ریسک متوسط، علامت‌دار، زیر ۳۰ سال}
\item کاربرد اولیه آن افتراق کیست از توده است.
\item در بیماران با ریسک بالا و بیماران با برست دنس، سونوگرافی به همراه ماموگرافی از ماموگرافی به تنهایی بهتر است.
\end{BIpoints}

\BInote{۹}{\en{MRI}}
\begin{BIpoints}
\item غربالگری به‌خصوص بیماران \en{high-risk}
\item ارزیابی بیماران با ایمپلنت برست
\item ارزیابی پاسخ به \en{ChT} نئوادجوانت
\item بیماران با لنف نود درگیر ولی محل تومور اولیه ناشناخته
\end{BIpoints}
\BInote{۱۰}{\en{MRI} حساس‌ترین تصویربرداری برست است.}
\BInote{۱۱}{\textbf{\en{MRI} استاندارد طلایی بررسی اینتگریتی ساختار ایمپلنت‌های سیلیکونی برست است.}}
\BInote{۱۲}{تقریباً همه \en{MRI}های برست با کنتراست گادولینیوم انجام می‌شوند.}
\BInote{۱۳}{ماموگرافی غربالگری}
\begin{BIpoints}
\item حساسیت آن وابسته به دانسیته برست است.
\item \en{Almost entirely fatty $\longrightarrow$ Very dense}
\item شامل ۲ گرافی از هر برست:
\begin{itemize}
\item کرانیوکودال (\en{CC})
\item مدیولترال ابلیک (\en{MLO})
\end{itemize}
\end{BIpoints}
\BInote{۱۴}{\en{Screening}}
\begin{BIenglish}
\item RF: Average $\longrightarrow$ Age $\geq40$ $\longrightarrow$ Annual screening
\item RF: BRCA1,2 or untested with relative of BRCA1,2 $\longrightarrow$ Age $\geq30$ $\longrightarrow$ Annual screening
\item RF: History of radiation (mantle, for Hodgkin) received between the age 10--30
\begin{itemize}
\item Starting 8 years after the radiation but not before age 25
\item Annual screening
\end{itemize}
\end{BIenglish}
\BIfoot{۳}
% pic04.jpg
\BIpage{۴}
\begin{BIenglish}
\item RF: $\geq20\%$ lifetime risk for breast cancer based on FH
\begin{itemize}
\item Age $\geq30$ or 10 years earlier than the age of diagnosis of the youngest affected relative, whichever is later
\item Annual screening
\end{itemize}
\item RF: Mother or sister with premenopausal breast cancer
\begin{itemize}
\item Age $\geq30$ or 10 years earlier, whichever is later
\item Annual screening
\end{itemize}
\item RF: Biopsy proven lobular neoplasia, Atypical ductal hyperplasia, DCIS, Invasive breast cancer, ovarian cancer
\begin{itemize}
\item From time of diagnosis regardless the age
\item Annual screening
\end{itemize}
\end{BIenglish}

\BInote{۱۶}{توصیه می‌شود توقف غربالگری براساس شرایط بالینی بیمار باشد نه سن بیمار.}
\BInote{۱۷}{ماموگرافی تشخیصی}
\begin{BIpoints}
\item در زنان علامت‌دار انجام می‌شود.
\item در زنانی که به علت غربالگری غیرطبیعی تحت بررسی بیشتر قرار می‌گیرند.
\item زنان با سابقه کنسر برست، یا ایمپلنت برست
\end{BIpoints}
\BInote{۱۸}{هر بیمار با علامت برست و سن ۳۰ و بالاتر: در ابتدا ماموگرافی غربالگری.}
\BInote{۱۹}{هر بیمار با علامت برست و سن ۲۹ و زیر ۳۰: در ابتدا \en{US}؛ اگر یافته مشکوک: ماموگرافی.}
\BInote{۲۰}{ده درصد کنسرهای برست هیچ یافته تصویربرداری در ماموگرافی یا \en{US} ندارند.}
\begin{BIpoints}

\BInote{۲۱}{نئوپلاسم‌های برست}
\begin{BIpoints}
\item غیرمهاجم (۲۵\%)
\begin{itemize}
\item \en{DCIS} (۸۰\%)
\begin{itemize}
\item در سال ۱\% تهاجمی می‌شود.
\item معمولاً یک‌طرفه
\item معمولاً توده قابل لمس ایجاد نمی‌کند.
\end{itemize}
\item \en{LCIS} (۲۰\%)
\begin{itemize}
\item در ماموگرافی دیده نمی‌شود.
\item بافت برست را به هم نمی‌ریزد.
\item می‌تواند دوطرفه باشد.
\item اغلب قابل لمس نیست.
\end{itemize}
\end{itemize}
\item مهاجم (۷۵\%)
\begin{itemize}
\item \en{IDC} (۸۰\%)
\begin{itemize}
\item از \en{DCIS} یا \en{LCIS} منشأ می‌گیرد.
\item فیبروز و توده قابل لمس می‌دهد.
\item متاستاز: خون و لنف
\end{itemize}
\item \en{ILC} (۱۰\%)
\begin{itemize}
\item از \en{LCIS} منشأ می‌گیرد.
\item فیبروز جزئی، معمولاً لمس نمی‌شود.
\item متاستاز به \en{GI}، تخمدان، رحم
\end{itemize}
\end{itemize}
\item سرطان التهابی برست (\en{IBC})
\begin{itemize}
\item فرم ناشایع کارسینوم تهاجمی
\item در زنان جوان شایع‌تر است.
\item در استیج اولیه شبیه ماستیت می‌شود.
\item بسیار اگرسیو
\item تهاجم از لنف نود شایع است.
\item در نیمی از موارد دوطرفه است.
\end{itemize}
\end{BIpoints}

\begin{BIpoints}
\item سرطان مدولاری برست (\en{MBC})
\begin{itemize}
\item فرم ناشایع کارسینوم مهاجم
\item مرتبط با \en{BRCA} موتاسیون
\item شایع در زنان ژاپنی
\item زنان زیر ۵۰ سال را درگیر می‌کند.
\item توده قابل لمس ایجاد می‌کند.
\item معمولاً یک‌طرفه
\item پروگنوز خوب
\end{itemize}
\item توبولار کارسینوما
\begin{itemize}
\item نوعی سرطان مهاجم پستان
\item معمولاً در زنان بالای ۵۰ سال
\item پروگنوز بسیار خوب
\item \en{Smallest spiculated mass on mammo}
\end{itemize}
\end{BIpoints}

\BInote{۵}{بیوپسی استریوتاکتیک}
\begin{BIpoints}
\item به طور عمده در مورد کلسیفیکاسیون‌ها انجام می‌شود.
\item نتیجه:
\begin{itemize}
\item \en{Concordant}: برگشت به برنامه عادی غربالگری

\item پاتولوژیست: ضایعه فیبروآدنوم است.

\end{itemize}
\end{BIpoints}
\BInote{۶}{توده پستان در بارداری}
\begin{BIpoints}

\item ماموگرافی اگر بیمار \en{$\geq30$}
\item \en{US} اگر بیمار \en{$<30$}

\BInote{۷}{ترشح از نیپل}

\item شایع‌ترین علت دیسچارج خونی: \en{Solitary papilloma}
\item بررسی: ماموگرافی + \en{US}
\end{BIpoints}
