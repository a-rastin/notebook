\section*{۹۳ ـ انتروکولیت نکروزان ـ \lr{NEC}}
\noindent\textbf{تصویر ۱ ـ صفحهٔ دست‌نویس ۱}\par
{\footnotesize\lr{Subject: NEC}\quad انتروکولیت نکروزان، ۹۳\par
\lr{Year: \rule{8mm}{0.2pt}\quad Month: \rule{8mm}{0.2pt}\quad Date: \rule{8mm}{0.2pt}}\par}

\begin{itemize}
\item شایع‌ترین اورژانس روده در نوزادان نارس است.
\item \textbf{نارس بودن} مهم‌ترین ریسک فاکتور برای \lr{NEC} است.
  \begin{itemize}
  \item \textbf{اکثراً زیر ۳۴ هفته.}
  \item حدود ۱۰\% نوزادان زیر \lr{1500 gr}.
  \end{itemize}
\item \lr{NEC} در نوزادان ترم ناشایع است.
\item در نوزادان نارس خصوصاً زیر \lr{34 w}، دستگاه گوارش نابالغ است؛ آغاز تغذیهٔ روده‌ای منجر به \lr{NEC} می‌شود.
\item ۱۰\% موارد \lr{NEC} تغذیهٔ روده‌ای را شروع نکرده‌اند.
\item اگر قرار است تغذیهٔ روده‌ای شروع شود، \lr{NEC} در موارد تغذیه با شیر انسان کمتر است. (\lr{!})
\end{itemize}

\subsection*{شواهد بالینی زودهنگام \lr{NEC}}
\begin{itemize}
\item دیستانسیون شکمی، عدم تحمل تغذیه، تهوع، \textbf{خونریزی رکتال}.
\item برخی موارد: اسهال.
\end{itemize}

\subsection*{تظاهرات بالینی بعد از پیشرفت \lr{NEC}}
\begin{itemize}
\item دیستانسیون شکمی شدید، \textbf{استفراغ صفراوی}، آسیت، لتارژی، اریتم دیوارهٔ شکم، ناپایداری دما، به‌همراه اپیزودهای آپنه/برادی‌کاردی، \textbf{\lr{DIC}}، شوک.
\item اگر \textbf{دیوارهٔ شکم آبی‌رنگ} شد: شک به پرفوراسیون.
\item \textbf{در \lr{NEC}، میزان \lr{WBC} اغلب \lr{\(\downarrow\)} است ولی می‌تواند \lr{\(\uparrow\)} هم باشد.}
\item \textbf{ترومبوسایتوپنی در \lr{NEC} شایع است.}
\end{itemize}

{\footnotesize\lr{Subject: NEC}\quad ۹۳\par

\subsection*{یافته‌های احتمالی دیگر}
\begin{itemize}
\item \textbf{اختلالات انعقادی}.
\item \textbf{اسیدوز متابولیک}.
\item اختلال الکترولیت.
\item هایپو و هایپرگلیسمی.
\end{itemize}

\begin{itemize}
\item پاتوژن اختصاصی با \lr{NEC} ارتباط ندارد.
  \begin{itemize}
  \item کشت شاید کمک‌کننده باشد ولی تشخیصی نیست.
  \end{itemize}
\item برای تشخیص \lr{NEC}، \textbf{تصویر‌برداری ضروری است}.
  \begin{itemize}
  \item زودرس‌ترین یافته: \textbf{ایلئوس روده}؛ افزایش ضخامت جدار روده.
  \item \textbf{سطح مایع ـ هوا}.
  \item یافتهٔ پاتوگنومونیک: \textbf{پنوماتوزیس اینتستینالیس}؛ ناشی از \lr{\(H_2\)} حاصل از باکتری‌ها.
  \item \textbf{\lr{Intrahepatic venous gas}}.
  \item \textbf{پنوموپریتونئوم}: در پی پرفوراسیون.
  \end{itemize}
\item عدم وجود پنوماتوزیس اینتستینالیس، \lr{NEC} را \lr{R/O} نمی‌کند.
  \begin{itemize}
  \item در انتروکولیت هیرشپرونگ و گاستروانتریت شدید نیز می‌تواند دیده شود.
  \end{itemize}
\end{itemize}

\subsection*{درمان \lr{NEC}}
\begin{itemize}
\item \textbf{قطع تغذیهٔ روده‌ای}، \lr{NGT} و ساکشن، جایگزینی آب و الکترولیت، \lr{TPN}، \textbf{\lr{AB} وسیع‌الطیف}.
\item حدود ۵۰\% بیماران نیاز به جراحی دارند.
\item \textbf{پنوموپریتونئوم = \lr{OR}}.
\end{itemize}
