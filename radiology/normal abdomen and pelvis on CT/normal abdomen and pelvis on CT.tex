\section*{Normal abdomen and pelvis on CT:}
\end{latin}

\noindent ۱- در بیماران با \en{$\mathrm{Cr}>1.5\,\mathrm{mg/dl}$}
\NAPCTarrow کنتراست ید منجر به \en{ATN} می‌شود.
\begin{itemize}
  \item \en{DM}، \en{MM}، دهیدریشن
  \item غالباً برگشت‌پذیر است.
\end{itemize}

\noindent ۲- \en{ATN} در نتیجه کنتراست ید وابسته به دوز است.
\par\bigskip

\noindent ۳- افراد مبتلا به آسم ریسک بالاتری برای حساسیت به ماده کنتراست دارند.
\begin{itemize}
  \item استروئید، دیفن هیدرامین، سایمتیدین
\end{itemize}

\noindent ۴- در \en{CT} ها شکم کنتراست خوراکی می‌دهیم بجز:
\begin{itemize}
  \item تروما \NAPCTarrow باریوم سولفات
  \item سنگ
  \item آسیب عروق
\end{itemize}

\noindent ۵- اگر شک به پرفوریشن داشته باشیم از ماده خوراکی گاستروگرافین استفاده می‌کنیم.
\par\bigskip

\noindent ۶- اندیکاسیون‌ها کنتراست خوراکی برای \en{CT} شکم و لگن
\begin{itemize}
  \item اغلب کیس‌های درد شکم غیرتروماتیک
  \item \en{IBD}
  \item آسیب شکم و لگن
  \item مشخص کردن محل پرفوریشن و فیستول
\end{itemize}
\NAPCTfoot{۱}

\noindent ۷- اندیکاسیون کنتراست وریدی
\begin{itemize}
  \item شکم و لگن
    \begin{itemize}
      \item ارزیابی وجود توده بدخیم
      \item تروما
      \item تشخیص آنوریسم و دایسکشن آئورت
      \item درد شکم (آپاندیسیت)
    \end{itemize}
  \item قفسه سینه
    \begin{itemize}
      \item \en{CT} آنژیوگرافی ریه (\en{CT-PA}) برای آمبولی ریه
      \item بررسی توده یا آدنوپاتی مدیاستن
      \item تشخیص آنوریسم و دایسکشن آئورت
      \item تشخیص بیماری‌های پلور (بدخیمی، آمپیم)
      \item ارزیابی شریان‌های کرونری
    \end{itemize}
\end{itemize}

\noindent ۸- اندیکاسیون عدم استفاده از کنتراست وریدی
\begin{itemize}

    \end{itemize}
  \item قفسه سینه
    \begin{itemize}
      \item \en{HRCT}
      \item تایید ندول ریوی دیده شده در گرافی ساده
      \item بررسی پنوموتوراکس و پنومومدیاستن
      \item کلسیم اسکورینگ شریان‌ها کرونری
      \item حساسیت به کنتراست یا نارسایی کلیوی
    \end{itemize}
\end{itemize}
